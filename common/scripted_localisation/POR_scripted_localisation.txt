# Renders the current Portuguese sphere in the Solor and Timor archipelago.
defined_text = {
	name = GetPORTimorSphere
	text = {
		trigger = { check_variable = { POR.POR_timor_influence > 79 } }
		localization_key = POR_timor_sphere_5
	}
	text = {
		trigger = { check_variable = { POR.POR_timor_influence > 69 } }
		localization_key = POR_timor_sphere_4
	}
	text = {
		trigger = { check_variable = { POR.POR_timor_influence > 54 } }
		localization_key = POR_timor_sphere_3
	}
	text = {
		trigger = { check_variable = { POR.POR_timor_influence > 29 } }
		localization_key = POR_timor_sphere_1
	}
	text = { localization_key = POR_timor_sphere_0 }
}

# Who currently holds the initiative in the archipelago.
defined_text = {
	name = GetPORTimorInitiative
	text = {
		trigger = { check_variable = { POR.POR_timor_influence > 50 } }
		localization_key = POR_timor_initiative_por
	}
	text = {
		trigger = { check_variable = { POR.POR_timor_influence < 50 } }
		localization_key = POR_timor_initiative_hol
	}
	text = { localization_key = POR_timor_initiative_none }
}

# Status line for each island group on the map, read from who actually owns the
# state: Portugal, the client Kingdom of Sunda (Dutch sphere) or Batavia directly.
defined_text = {
	name = GetPORTimorEastTimor
	text = { trigger = { 721 = { is_owned_by = POR } } localization_key = POR_timor_area_portuguese }
	text = { localization_key = POR_timor_area_dutch }
}
