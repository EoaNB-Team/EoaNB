	open_parliament_window = {
		context_type = player_context

		window_name = "open_parliament_window"
		parent_window_token = politics_tab

		visible = {
			is_ai = no
			has_generic_parliament = yes
		}

		effects = {
			open_button_click = {
				if = {
					limit = { has_variable = show_parliament_window }

			}
		}

		ai_enabled = { always = no }
	}

	parliament_window = {
		context_type = player_context

		window_name = "parliament_window"

		# Rebuilding 256 seat icons every tick is the single most expensive thing this GUI does.
		# parliament_seats_dirty is bumped by compute_seats, draw_seats, set_global_seats_posistions,
		# parliament_recompute_coalition_seats and by every parliament button, so the chamber
		# refreshes whenever it can actually have changed. Delete this one line to go back to
		# updating every tick.
		dirty = parliament_seats_dirty

		visible = { check_variable = { show_parliament_window > 0 } }

		properties = {
			parliament_dot_icon = {
				x = global.internal_politics_house_x^seat_idx
				y = global.internal_politics_house_y^seat_idx
				frame = seats_array^seat_idx
			}
			# Government support bar, 0-100. parliament_support_percent is maintained by
			# parliament_recompute_coalition_seats.
			parliament_support_bar = { frame = parliament_support_percent }

			# Tab highlight.
			parties_button  = { frame = parl_tab_parties_frame }
			decision_button = { frame = parl_tab_decisions_frame }
			reform_button   = { frame = parl_tab_reforms_frame }
		}

		dynamic_lists = {
			parliament_diagram = {
				array = global.internal_politics_house_x
				change_scope = no
				entry_container = internal_politics_house_seat
