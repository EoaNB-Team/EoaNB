FRA_imperial_influence_inlay_window = {

	internal = yes

	visible = { has_country_flag = FRA_imperial_influence_active }

	scripted_buttons = {
		FRA_conservative_bonapartists_endorse = {
			available = {
				custom_trigger_tooltip = {
					tooltip = FRA_imperial_influence_endorse_cost_available_tt
					OR = {
						AND = { FRA_conservative_bonapartists_are_subdued = yes has_political_power > 199 }
						AND = { NOT = { FRA_conservative_bonapartists_are_subdued = yes } has_political_power > 74 }
					}
				}
				custom_trigger_tooltip = { tooltip = FRA_imperial_influence_below_dominating_tt check_variable = { FRA_conservative_bonapartists_influence < 4 } }
				custom_trigger_tooltip = { tooltip = FRA_imperial_influence_intervention_ready_tt NOT = { has_country_flag = FRA_intervened_in_imperial_influence } }
			}
			click_effect = {
				if = { limit = { FRA_conservative_bonapartists_are_subdued = yes } add_political_power = -200 }
				else = { add_political_power = -75 }
				FRA_increase_conservative_bonapartist_influence = yes
				set_country_flag = { flag = FRA_intervened_in_imperial_influence value = 1 days = 180 }
			}
		}
		FRA_conservative_bonapartists_discredit = {
