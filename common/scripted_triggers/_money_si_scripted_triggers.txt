# Input:  si_in
# Output: si_out (scaled), si_tier (0 = K, 1 = M, 2 = B)
format_money_si = {
	set_temp_variable = { si_out = si_in }
	set_temp_variable = { si_tier = 1 }

	set_temp_variable = { si_abs = si_in }
	if = {
		limit = { check_variable = { si_abs < 0 } }
		multiply_temp_variable = { si_abs = -1 }
	}

	if = {
		# billions
		limit = { check_variable = { si_abs > 999 } }
		divide_temp_variable = { si_out = 1000 }
		set_temp_variable = { si_tier = 2 }
	}
	else_if = {
		limit = {
			check_variable = { si_abs < 1 }
			check_variable = { si_abs > 0.0001 }
		}
		multiply_temp_variable = { si_out = 1000 }
		set_temp_variable = { si_tier = 0 }
	}
	set_temp_variable = { si_abs = si_out }
	if = {
		limit = { check_variable = { si_abs < 0 } }
		multiply_temp_variable = { si_abs = -1 }
	}
	set_temp_variable = { si_dec = 2 }
	if = {
		limit = { check_variable = { si_abs > 99.999 } }
		set_temp_variable = { si_dec = 0 }
	}
	else_if = {
		limit = { check_variable = { si_abs > 9.999 } }
		set_temp_variable = { si_dec = 1 }
	}
}
