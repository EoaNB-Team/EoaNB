	OR = {
		AND = {
			NOT = { is_owned_by = UCT }
			OWNER = { is_ai = no }
		}
		AND = {
			NOT = { is_owned_by = UCT }
			OWNER = {
				is_subject = yes
				OVERLORD = { is_ai = no }
			}
		}
		any_country = {
			is_ai = no
			NOT = { tag = event_target:scramble_recipient }
			PREV = { is_claimed_by = PREV }
		}
	}
}
