# De Morny dies on 10.03.1865; his branch only resumes if a successor takes his seat.

	custom_trigger_tooltip = {
		tooltip = FRA_ministry_de_morny_in_office_tt
		has_country_flag = FRA_ministry_member_de_morny
		OR = {
			NOT = { has_country_flag = FRA_ministry_de_morny_departed }
			has_country_flag = FRA_ministry_ollivier_succeeded_de_morny
		}
	}
}
