convert_bar_scratch_to_frame = {
	if = { limit = { check_variable = { bar_scratch < 0 } } set_temp_variable = { bar_scratch = 0 } }
	if = {
		limit = { check_variable = { bar_denom > 0.001 } }
		divide_temp_variable = { bar_scratch = bar_denom }
	}
	else = { set_temp_variable = { bar_scratch = 0 } }
	if = { limit = { check_variable = { bar_scratch > 1 } } set_temp_variable = { bar_scratch = 1 } }
	multiply_temp_variable = { bar_scratch = 20 }
	round_temp_variable = bar_scratch
	add_to_temp_variable = { bar_scratch = 1 }
}

# Months of treasury left at the current balance. 0 when running a surplus.
calculate_treasury_runway = {
	set_temp_variable = { treasury_runway_months = 0 }
	if = {
		limit = { check_variable = { total_balance_temp < -0.001 } }
		set_temp_variable = { treasury_runway_months = money }
		set_temp_variable = { bar_denom = total_balance_temp }
		multiply_temp_variable = { bar_denom = -1 }
		divide_temp_variable = { treasury_runway_months = bar_denom }
		round_temp_variable = treasury_runway_months
		if = { limit = { check_variable = { treasury_runway_months < 0 } } set_temp_variable = { treasury_runway_months = 0 } }
	}
}
