# Has at least `gg_points_to_check` Great Game points.
# Usage:
#   custom_cost_trigger = {
#       set_temp_variable = { gg_points_to_check = 5 }
#       gg_has_points = yes
#   }
#   custom_cost_text = great_game_points_cost
gg_has_points = {
	check_variable = {
		var = great_game_points
		value = gg_points_to_check
		compare = greater_than_or_equals
	}
}

gg_sphere_is_open = {
	exists = yes
	NOT = {
		has_country_flag = RUS_sphere
		has_country_flag = ENG_sphere
	}
	NOT = { has_global_flag = great_game_anglo_russian_convention }
}

gg_sphere_rival_is_dominating = {
	OR = {
		AND = {
			ROOT = { tag = ENG }
			has_country_flag = RUS_becoming_dominate
		}
		AND = {
			ROOT = { tag = RUS }
			has_country_flag = ENG_becoming_dominate
		}
	}
}
