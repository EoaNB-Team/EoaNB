# Victorian 1854-1900 project: on_actions.
# on_startup is merged with the other on_startup blocks of the mod; only a single effect is run here.
on_actions = {
	on_startup = {
		effect = {
			# on_startup has no scope (None); scripted effects need a country scope, so run once in ENG.
			if = {
				limit = { has_start_date < 1858.1.1 }
				every_country = {
					limit = { original_tag = ENG }
					v54_initialise = yes
				}
				# The Russo-Turkish war is under way on 1 January 1854 (hidden event, scope Russia).
				every_country = {
					limit = { original_tag = RUS }
					country_event = { id = v54_eur.1 days = 2 }
				}
			}
		}
	}
}
