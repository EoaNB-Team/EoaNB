on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { NOT = { has_country_flag = eoanb_literacy_initialised } }
				eoanb_literacy_init = yes
			}
		}
	}

	on_monthly = {
		effect = {
			eoanb_literacy_monthly_tick = yes

			if = {
				limit = { NOT = { check_variable = { global.eoanb_literacy_map_refresh_day = global.num_days } } }
				set_variable = { global.eoanb_literacy_map_refresh_day = global.num_days }
				force_update_map_mode = { mapmode = eoanb_literacy_map_mode }
			}
		}
	}

	on_release_as_free = {
		effect = {
			ROOT = {
				if = {
					limit = { NOT = { has_country_flag = eoanb_literacy_initialised } }
					eoanb_literacy_init = yes
				}
			}
		}
	}

	on_release_as_puppet = {
		effect = {
			ROOT = {
				if = {
					limit = { NOT = { has_country_flag = eoanb_literacy_initialised } }
					eoanb_literacy_init = yes
				}
			}
		}
	}

	on_liberate = {
		effect = {
			ROOT = {
				if = {
					limit = { NOT = { has_country_flag = eoanb_literacy_initialised } }
					eoanb_literacy_init = yes
				}
			}
		}
	}

	on_declare_war = {
		effect = {
			if = {
				limit = {
					ROOT = { NOT = { has_country_flag = eoanb_literacy_initialised } }
					FROM = { has_country_flag = eoanb_literacy_initialised }
					OR = {
						ROOT = { original_tag = FROM }
						FROM = { original_tag = ROOT }
					}
				}
				FROM = { ROOT = { eoanb_literacy_inherit_from_prev = yes } }
			}
			else_if = {
				limit = {
					FROM = { NOT = { has_country_flag = eoanb_literacy_initialised } }
					ROOT = { has_country_flag = eoanb_literacy_initialised }
					OR = {
						ROOT = { original_tag = FROM }
						FROM = { original_tag = ROOT }
					}
				}
				ROOT = { FROM = { eoanb_literacy_inherit_from_prev = yes } }
			}
		}
	}

	on_annex = {
		effect = {
			ROOT = { eoanb_literacy_absorb_from = yes }
		}
	}
}
