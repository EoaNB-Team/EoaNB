on_actions = {

	on_startup = { effect = { UCT = { EOANBSYS_pandemics_constants_setup = yes } } }

	on_weekly = {
		effect = {
			if = {
				limit = { EOANBSYS_influenza_is_relevant_country = yes }
				EOANBSYS_influenza_weekly_tick = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { EOANBSYS_influenza_is_relevant_country = yes }
				EOANBSYS_influenza_refresh_spread_factors = yes
			}
		}
	}
}
