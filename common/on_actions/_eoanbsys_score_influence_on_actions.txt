on_actions = {
	# Setup
	on_startup = {
		effect = {
			# Must run before the first calc - history files only create
			# prestige_score for the handful of nations that start with one
			every_country = { eoanb_score_vars_init = yes }

			UCT = {
				set_variable = { global.time_counter = 3 }
				eoanb_score_system_calc = yes
				eoanb_ledger_system_calc = yes
				eoanb_score_system_calc = yes
				eoanb_ledger_system_calc = yes
				eoanb_constants_setup = yes
			}
			every_country = {
				EOANBSYS_institutions_era_initialize = yes
				if = {
					limit = { has_variable = rival_country }
					meta_effect = {
						text = {
							add_opinion_modifier = { 
								target = [RIVAL]
								modifier = rival_country
							}
						}
						RIVAL = "[?rival_country.GetTag]"
					}  
				}
				if = {
					limit = { has_variable = most_favoured_country }
					meta_effect = {
						text = {
							add_opinion_modifier = {
								target = [MOST_FAVOURED_COUNTRY]
								modifier = most_favoured_country
							}
						}
						MOST_FAVOURED_COUNTRY = "[?most_favoured_country.GetTag]"
					}					
				}
			}
		}
	} 

	# Update
	on_monthly_UCT = {
		effect = {
			add_to_variable = { global.time_counter = 1 }
			if = {
				limit = { check_variable = { global.time_counter > 3 } }
				set_variable = { global.time_counter = 0 }			
				eoanb_score_system_calc = yes 
				eoanb_ledger_system_calc = yes 							
			}
		}
	}
	
	# ROOT is winner #FROM gets annexed - For civil wars on_civil_war_end is also fired
	on_annex = {
		effect = {
			FROM = { eoanb_score_vars_init = yes }
			ROOT = {
				eoanb_score_vars_init = yes
				# winner absorbs half of the annexed nation's modified scores
				add_to_variable = { modified_military_score   = { value = FROM.modified_military_score   multiply = 0.50 } }
				add_to_variable = { modified_industrial_score = { value = FROM.modified_industrial_score multiply = 0.50 } }
			}
		}
	}
	
	# FROM is the one that joins the faction
	on_create_faction = {
		effect = {
			FROM = {
				eoanb_score_vars_init = yes
				add_to_variable = { modified_military_score = 25 }
			}
		}
	}
	
	# NOTE: direction kept as in the original script (FROM gains, ROOT loses).
	# Swap the two multipliers if testing shows the scopes are the other way round.
	on_peaceconference_ended = {
		effect = {
			FROM = {
				eoanb_score_vars_init = yes
				multiply_variable = { modified_military_score = 1.20 }
			}
			ROOT = {
				eoanb_score_vars_init = yes
				multiply_variable = { modified_military_score = 0.80 }
			}
		}
	}
}
