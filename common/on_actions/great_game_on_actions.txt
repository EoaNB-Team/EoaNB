on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { has_start_date < 1904.1.1 }
				random_country = {
					limit = {
						OR = {
							tag = RUS
							tag = ENG
						}
					}
					great_game_setup = yes
				}

				if = {
					limit = {
						is_historical_focus_on = yes
						QNG = { is_ai = yes }
						TAI = {	is_ai = yes }
					}
					QNG = { add_ideas = QNG_idea_hidden_tps_whineing }
				}
			}
		}
	}

	on_monthly_RUS = {
		effect = {
			if = {
				limit = { has_global_flag = EDC_great_game_progress }
				gg_update_stage = yes
			}
		}
	}
	on_monthly_ENG = {
		effect = {
			if = {
				limit = { has_global_flag = EDC_great_game_progress }
				gg_update_stage = yes
			}
		}
	}
}
