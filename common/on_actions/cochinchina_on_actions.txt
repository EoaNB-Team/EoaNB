###########################################################################
#	The Cochinchina Expedition - On Actions
###########################################################################

on_actions = {

	on_startup = {
		effect = {
			VIN = {
				VIN_Western_Influence_Drift_Start = yes

				set_country_flag = VIN_Opium_Tab_Open

				VIN_Agrarian_Roots_Seed_Variables = yes

				set_power_balance = {
					id = VIN_western_influence_bop
					left_side = VIN_traditionalist_side
					right_side = VIN_western_side
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					ROOT = {
						TAG = VIN
						has_country_flag = VIN_cochinchina_intervention_flag
					}
					NOT = { has_global_flag = SPR_cochinchina_expedition_capitulation_flag }
					any_country = {
						cochinchina_expedition_is_participant_trigger = yes
						has_war_with = VIN
					}
				}
				cochinchina_expedition_capitulation_effect = yes
			}
		}
	}

	on_annex = {
		effect = {
			if = {
				limit = {
					FROM = { TAG = VIN }
					has_global_flag = SPR_cochinchina_intervention_global_flag
					NOT = { has_global_flag = SPR_cochinchina_expedition_capitulation_flag }
				}
				cochinchina_expedition_abort_effect = yes
				cochinchina_expedition_close_joint_access_effect = yes
				cochinchina_expedition_withdraw_expeditionary_force_effect = yes
			}
		}
	}
}
