on_actions = {
	on_weekly = {
		effect = {
			# UNEMPLOYMENT SYSTEM #
			set_temp_variable = { unemployment_value = modifier@weekly_unemployment_change }
			if = {
				limit = {
					check_variable = { unemployment_value > 0 }
					check_variable = {
						var = current_unemployment
						value = 1
						compare = greater_than_or_equals
					}
				}
				add_unemployment_effect = yes
			}
			building_bonus_weekly_effect = yes
		}
	}
	on_weekly_AUS = {
		effect = {
			if = {
				limit = {
					NOT = { has_variable = civ_total_bonus }
					OR = {
						has_variable = AUS_var_wien_urbanisation
						has_variable = AUS_var_bohemian_industrialization
					}
				}
				clear_variable = AUS_var_wien_urbanisation
				clear_variable = AUS_var_bohemian_industrialization
			}
		}
	}
	on_weekly_POR = {
		effect = {
			# TIMOR INFLUENCE CONTEST #
			# Whoever already dominates the Solor and Timor archipelago slowly consolidates.
			if = {
				limit = {
					has_country_flag = POR_timor_influence_active
					NOT = { has_country_flag = POR_timor_governance_settled }
				}
				if = {
					limit = { check_variable = { POR_timor_influence > 50 } }
					add_to_variable = { POR_timor_influence = 0.1 }
				}
				else_if = {
					limit = { check_variable = { POR_timor_influence < 50 } }
					subtract_from_variable = { POR_timor_influence = 0.1 }
				}
				POR_timor_influence_recalc_effect = yes
			}
		}
	}
	on_weekly_FRA = {
		effect = {
			if = {
				limit = {
					has_completed_focus = FRA_pour_dieu_et_le_roi
					NOT = { has_country_flag = fra_orsini_failure }
					NOT = { has_country_flag = fra_orsini_success }
				}
				add_to_variable = { FRA_royal_support = FRA_royal_support_weekly }
				clamp_variable = { var = FRA_royal_support min = 0 max = 9999 }
			}
		}
	}
}
