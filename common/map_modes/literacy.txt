scripted_map_modes = {
	eoanb_literacy_map_mode = {
		top = {
			type = country
			color = {
				set_temp_variable = { literacy_shade = FROM.literacy_rate }
				divide_temp_variable = { literacy_shade = 100 }
				clamp_temp_variable = { var = literacy_shade min = 0 max = 1 }

				set_temp_variable = { red = literacy_shade }
				multiply_temp_variable = { red = 0.5 }
				add_to_temp_variable = { red = 0.4 }

				set_temp_variable = { green = literacy_shade }
				multiply_temp_variable = { green = 0.8 }
				add_to_temp_variable = { green = 0.1 }

				set_temp_variable = { blue = literacy_shade }
				multiply_temp_variable = { blue = 0.5 }
				add_to_temp_variable = { blue = 0.1 }

				set_temp_variable = { alpha = 1 }
			}
			thickness = no
		}
		bottom = { type = none }
		far_text = country
		near_text = state
		update_daily = no
	}
}
