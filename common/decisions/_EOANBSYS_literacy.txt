eoanb_category_institutions = {
	eoanb_decision_literacy_campaign = {
		icon = education

		visible = {
			has_country_flag = eoanb_literacy_initialised
			check_variable = { literacy_rate < 90 }
			NOT = { has_idea = oral_tradition }
		}

		available = {
			NOT = { has_idea = bankruptcy }
			has_war = no
		}

		days_remove = 180
		days_re_enable = 550

		custom_cost_trigger = {
			set_temp_variable = { amount_of_money_to_check = num_of_factories }
			multiply_temp_variable = { amount_of_money_to_check = 0.15 }
			if = {
				limit = { check_variable = { amount_of_money_to_check > 10000 } }
				set_temp_variable = { amount_of_money_to_check = 10000 }
			}
			has_specific_amount_of_money_with_inflation_included = yes
		}
		custom_cost_text = eoanb_money_cost_text

		modifier = {
			political_power_gain = -0.1
			research_spending_cost_factor = 0.1
			hidden_modifier = {
				research_spending_cost_factor_from_decisions = 0.1
			}
		}

		complete_effect = {
			set_temp_variable = { money_to_gain = num_of_factories }
			multiply_temp_variable = { money_to_gain = 0.15 }
			if = {
				limit = { check_variable = { money_to_gain > 10000 } }
				set_temp_variable = { money_to_gain = 10000 }
			}
			multiply_temp_variable = { money_to_gain = -1 }
			add_money_with_inflation_included_with_tooltip_effect = yes
		}

		remove_effect = {
			eoanb_add_2_literacy = yes
		}

		ai_will_do = {
			base = 1
			modifier = {
				factor = 2
				check_variable = { literacy_rate < 45 }
			}
		}
	}

	eoanb_decision_literacy_teacher_seminaries = {
		icon = university

		visible = {
			has_country_flag = eoanb_literacy_initialised
			check_variable = { literacy_rate < 90 }
			OR = {
				has_idea = private_education
				has_idea = public_schooling_system
			}
		}

		available = {
			NOT = { has_idea = bankruptcy }
			num_of_civilian_factories_available_for_projects > 0
		}

		fire_only_once = yes

		days_remove = 365

		custom_cost_trigger = {
			set_temp_variable = { amount_of_money_to_check = num_of_factories }
			multiply_temp_variable = { amount_of_money_to_check = 0.3 }
			if = {
				limit = { check_variable = { amount_of_money_to_check > 10000 } }
				set_temp_variable = { amount_of_money_to_check = 10000 }
			}
			has_specific_amount_of_money_with_inflation_included = yes
		}
		custom_cost_text = eoanb_money_cost_text

		modifier = {
			civilian_factory_use = 1
		}

		complete_effect = {
			set_temp_variable = { money_to_gain = num_of_factories }
			multiply_temp_variable = { money_to_gain = 0.3 }
			if = {
				limit = { check_variable = { money_to_gain > 10000 } }
				set_temp_variable = { money_to_gain = 10000 }
			}
			multiply_temp_variable = { money_to_gain = -1 }
			add_money_with_inflation_included_with_tooltip_effect = yes
		}

		remove_effect = {
			set_temp_variable = { literacy_growth_change = 0.15 }
			eoanb_literacy_growth_change = yes
		}

		ai_will_do = { base = 1 }
	}

	eoanb_decision_literacy_regimental_schools = {
		icon = newspaper_open

		visible = {
			has_country_flag = eoanb_literacy_initialised
			check_variable = { literacy_rate < 70 }
			eoanb_literacy_has_conscript_army = yes
		}

		available = {
			has_war = no
			has_army_experience > 14.99
		}

		cost = 25
		days_remove = 120
		days_re_enable = 730

		modifier = {
			training_time_factor = 0.05
		}

		complete_effect = {
			army_experience = -15
		}

		remove_effect = {
			eoanb_add_1_literacy = yes
		}

		ai_will_do = { base = 1 }
	}
}
