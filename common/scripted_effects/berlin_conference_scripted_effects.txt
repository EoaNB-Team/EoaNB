scramble_africa_grant_state = {
	if = {
		limit = {
			NOT = { is_owned_by = event_target:scramble_recipient }
			NOT = { OWNER = { is_subject_of = event_target:scramble_recipient } }
			OR = {
				is_owned_by = UCT
				OWNER = { capital_scope = { is_on_continent = africa } }
			}
		}
		scramble_africa_grant_state_execute = yes
	}
}

scramble_africa_grant_state_allow_owner = {
	if = {
		limit = {
			NOT = { is_owned_by = event_target:scramble_recipient }
			NOT = { OWNER = { is_subject_of = event_target:scramble_recipient } }
			OR = {
				is_owned_by = UCT
				is_owned_by = event_target:scramble_allowed_owner
				OWNER = { capital_scope = { is_on_continent = africa } }
			}
		}
		scramble_africa_grant_state_execute = yes
	}
}

scramble_africa_grant_state_execute = {
	if = {
		limit = { scramble_africa_state_is_player_protected = yes }
		add_claim_by = event_target:scramble_recipient
		scramble_africa_notify_players = yes
	}
	else = {
		event_target:scramble_recipient = { transfer_state = PREV }
	}
}

scramble_africa_notify_players = {
	event_target:scramble_recipient = { save_global_event_target_as = scramble_last_claimant }
	if = {
		limit = {
			NOT = { is_owned_by = UCT }
			OWNER = { is_ai = no }
		}
		OWNER = { scramble_africa_send_notice = yes }
	}
	if = {
		limit = {
			NOT = { is_owned_by = UCT }
			OWNER = {
				is_subject = yes
				OVERLORD = { is_ai = no }
			}
		}
		OWNER = { OVERLORD = { scramble_africa_send_notice = yes } }
	}
	every_country = {
		limit = {
			is_ai = no
			NOT = { tag = event_target:scramble_recipient }
			PREV = { is_claimed_by = PREV }
		}
		scramble_africa_send_notice = yes
	}
}

scramble_africa_send_notice = {
	if = {
		limit = { NOT = { has_country_flag = scramble_africa_notice_sent } }
		set_country_flag = { flag = scramble_africa_notice_sent days = 3 }
		country_event = { id = scramble_africa.900 }
	}
}
