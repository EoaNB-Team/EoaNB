store_current_ideologies = {
	hidden_effect = {
		set_variable = { var_stored_ruling_party = current_party_ideology_group }
 
		### IDEOLOGY LIST ###
		set_variable = { var_stored_pop_@centrism = party_popularity_100@centrism }
		set_variable = { var_stored_pop_@ideology_social_liberalism = party_popularity_100@ideology_social_liberalism }
		set_variable = { var_stored_pop_@traditional_conservatism = party_popularity_100@traditional_conservatism }
		set_variable = { var_stored_pop_@autocracy = party_popularity_100@autocracy }
		set_variable = { var_stored_pop_@social_egalitarianism = party_popularity_100@social_egalitarianism }
		set_variable = { var_stored_pop_@ideology_radical_socialism = party_popularity_100@ideology_radical_socialism }
		set_variable = { var_stored_pop_@fundamentalism = party_popularity_100@fundamentalism }
		set_variable = { var_stored_pop_@chauvinist_populism = party_popularity_100@chauvinist_populism }
		set_variable = { var_stored_pop_@anarchism = party_popularity_100@anarchism }
		set_variable = { var_stored_pop_@radical_democracy = party_popularity_100@radical_democracy }
	}
}
 
load_prevous_ideologies = {
	hidden_effect = {
		if = {
			limit = { has_variable = var_stored_ruling_party }
 
			set_politics = { ruling_party = var:var_stored_ruling_party }
			set_temp_variable = { temp_popularity_sum = 0 }
 
			### IDEOLOGY LIST ###
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@centrism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@ideology_social_liberalism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@traditional_conservatism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@autocracy }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@social_egalitarianism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@ideology_radical_socialism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@fundamentalism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@chauvinist_populism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@anarchism }
			add_to_temp_variable = { temp_popularity_sum = var_stored_pop_@radical_democracy }
 
			set_temp_variable = { temp_popularity_correction = 100 }
			subtract_from_temp_variable = { temp_popularity_correction = temp_popularity_sum }
 
			add_to_variable = { var_stored_pop_@var:var_stored_ruling_party = temp_popularity_correction }
 
			set_popularities = {
				### IDEOLOGY LIST ###
				centrism                   = var:var_stored_pop_@centrism
				ideology_social_liberalism = var:var_stored_pop_@ideology_social_liberalism
				traditional_conservatism   = var:var_stored_pop_@traditional_conservatism
				autocracy                  = var:var_stored_pop_@autocracy
				social_egalitarianism      = var:var_stored_pop_@social_egalitarianism
				ideology_radical_socialism = var:var_stored_pop_@ideology_radical_socialism
				fundamentalism             = var:var_stored_pop_@fundamentalism
				chauvinist_populism        = var:var_stored_pop_@chauvinist_populism
				anarchism                  = var:var_stored_pop_@anarchism
				radical_democracy          = var:var_stored_pop_@radical_democracy
			}
 
			clear_variable = var_stored_ruling_party
 
			### IDEOLOGY LIST ###
			clear_variable = var_stored_pop_@centrism
			clear_variable = var_stored_pop_@ideology_social_liberalism
			clear_variable = var_stored_pop_@traditional_conservatism
			clear_variable = var_stored_pop_@autocracy
			clear_variable = var_stored_pop_@social_egalitarianism
			clear_variable = var_stored_pop_@ideology_radical_socialism
			clear_variable = var_stored_pop_@fundamentalism
			clear_variable = var_stored_pop_@chauvinist_populism
			clear_variable = var_stored_pop_@anarchism
			clear_variable = var_stored_pop_@radical_democracy
		}
	}
}
