ITA_2IW_remove_war_ideas = {
	hidden_effect = {
		if = {
			limit = { PIE = { has_idea = ITA_2IP_War_Leader } }
			PIE = { remove_ideas = ITA_2IP_War_Leader }
		}
		if = {
			limit = { FRA = { has_idea = ITA_2IP_War_Leader } }
			FRA = { remove_ideas = ITA_2IP_War_Leader }
		}
		if = {
			limit = { PIE = { has_idea = ITA_2IP_Idea } }
			PIE = { remove_ideas = ITA_2IP_Idea }
		}
		if = {
			limit = { FRA = { has_idea = ITA_2IP_Idea } }
			FRA = { remove_ideas = ITA_2IP_Idea }
		}
		if = {
			limit = { AUS = { has_idea = AUS_idea_italian_war } }
			AUS = { remove_ideas = AUS_idea_italian_war }
		}
	}
}
