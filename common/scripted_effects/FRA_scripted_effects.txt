	else_if = { limit = { has_idea = regulated_press } set_variable = { FRA_imperial_influence_press_law_rank = 3 } }
	else_if = { limit = { has_idea = free_press } set_variable = { FRA_imperial_influence_press_law_rank = 4 } }

	set_variable = { FRA_imperial_influence_suffrage_law_rank = 1 }
	if = { limit = { has_idea = weighted_wealth_voting } set_variable = { FRA_imperial_influence_suffrage_law_rank = 2 } }
	else_if = { limit = { has_idea = universal_male_suffrage } set_variable = { FRA_imperial_influence_suffrage_law_rank = 3 } }
	else_if = { limit = { has_idea = universal_suffrage } set_variable = { FRA_imperial_influence_suffrage_law_rank = 4 } }

	else_if = { limit = { has_idea = freedom_of_religion } set_variable = { FRA_imperial_influence_religious_law_rank = 3 } }
	else_if = { limit = { has_idea = state_religion } set_variable = { FRA_imperial_influence_religious_law_rank = 4 } }
	else_if = { limit = { has_idea = theocracy } set_variable = { FRA_imperial_influence_religious_law_rank = 5 } }
}

FRA_update_imperial_influence_from_laws = {
	set_temp_variable = { FRA_current_press_law_rank = 0 }
	if = { limit = { has_idea = state_controlled_press } set_temp_variable = { FRA_current_press_law_rank = 1 } }
	else_if = { limit = { has_idea = censored_press } set_temp_variable = { FRA_current_press_law_rank = 2 } }
	else_if = { limit = { has_idea = regulated_press } set_temp_variable = { FRA_current_press_law_rank = 3 } }
	else_if = { limit = { has_idea = free_press } set_temp_variable = { FRA_current_press_law_rank = 4 } }

	set_temp_variable = { FRA_current_suffrage_law_rank = 1 }
	if = { limit = { has_idea = weighted_wealth_voting } set_temp_variable = { FRA_current_suffrage_law_rank = 2 } }
	else_if = { limit = { has_idea = universal_male_suffrage } set_temp_variable = { FRA_current_suffrage_law_rank = 3 } }
	else_if = { limit = { has_idea = universal_suffrage } set_temp_variable = { FRA_current_suffrage_law_rank = 4 } }

	set_temp_variable = { FRA_current_religious_law_rank = 1 }
	if = { limit = { has_idea = total_separation } set_temp_variable = { FRA_current_religious_law_rank = 2 } }
	else_if = { limit = { has_idea = freedom_of_religion } set_temp_variable = { FRA_current_religious_law_rank = 3 } }
	else_if = { limit = { has_idea = state_religion } set_temp_variable = { FRA_current_religious_law_rank = 4 } }
	else_if = { limit = { has_idea = theocracy } set_temp_variable = { FRA_current_religious_law_rank = 5 } }

	if = {
		limit = { NOT = { has_variable = FRA_imperial_influence_press_law_rank } }
		set_variable = { FRA_imperial_influence_press_law_rank = FRA_current_press_law_rank }
	}
	else_if = {
		limit = { NOT = { check_variable = { FRA_imperial_influence_press_law_rank = FRA_current_press_law_rank } } }
		add_to_variable = { FRA_liberal_bonapartists_influence = FRA_current_press_law_rank }
		subtract_from_variable = { FRA_liberal_bonapartists_influence = FRA_imperial_influence_press_law_rank }
		set_variable = { FRA_imperial_influence_press_law_rank = FRA_current_press_law_rank }
		set_temp_variable = { FRA_imperial_influence_law_changed = 1 }
	}

	if = {
		limit = { NOT = { has_variable = FRA_imperial_influence_suffrage_law_rank } }
		set_variable = { FRA_imperial_influence_suffrage_law_rank = FRA_current_suffrage_law_rank }
	}
	else_if = {
		limit = { NOT = { check_variable = { FRA_imperial_influence_suffrage_law_rank = FRA_current_suffrage_law_rank } } }
		add_to_variable = { FRA_radical_bonapartists_influence = FRA_current_suffrage_law_rank }
		subtract_from_variable = { FRA_radical_bonapartists_influence = FRA_imperial_influence_suffrage_law_rank }
		set_variable = { FRA_imperial_influence_suffrage_law_rank = FRA_current_suffrage_law_rank }
		set_temp_variable = { FRA_imperial_influence_law_changed = 1 }
	}

	if = {
		limit = { NOT = { has_variable = FRA_imperial_influence_religious_law_rank } }
		set_variable = { FRA_imperial_influence_religious_law_rank = FRA_current_religious_law_rank }
	}
	else_if = {
		limit = { NOT = { check_variable = { FRA_imperial_influence_religious_law_rank = FRA_current_religious_law_rank } } }
		add_to_variable = { FRA_legitimist_bonapartists_influence = FRA_current_religious_law_rank }
		subtract_from_variable = { FRA_legitimist_bonapartists_influence = FRA_imperial_influence_religious_law_rank }
		set_variable = { FRA_imperial_influence_religious_law_rank = FRA_current_religious_law_rank }
		set_temp_variable = { FRA_imperial_influence_law_changed = 1 }
	}

	if = {
		limit = { check_variable = { FRA_imperial_influence_law_changed = 1 } }
		FRA_refresh_imperial_influence = yes
	}
}

# The Imperial Ministry (inner circle) - state handling
#
# Seat variables hold the identity of the sitting minister:
# 0 = vacant, 1 = Walewski, 2 = de Morny, 3 = Eugenie, 4 = Thouvenel,
# 5 = Rouher, 6 = Persigny, 7 = Fould

FRA_imperial_ministry_initialise = {
	custom_effect_tooltip = FRA_imperial_ministry_unlocked_tt
	hidden_effect = {
		set_variable = { FRA_ministry_seat_1 = 0 }
		set_variable = { FRA_ministry_seat_2 = 0 }
		set_variable = { FRA_ministry_seat_3 = 0 }
		set_variable = { FRA_ministry_seats_used = 0 }
		set_country_flag = FRA_imperial_ministry_active
	}
}

# Expects FRA_ministry_appointee to be set beforehand.
FRA_ministry_seat_assign = {
	if = {
		limit = { check_variable = { FRA_ministry_seat_1 = 0 } }
		set_variable = { FRA_ministry_seat_1 = FRA_ministry_appointee }
		add_to_variable = { FRA_ministry_seats_used = 1 }
	}
	else_if = {
		limit = { check_variable = { FRA_ministry_seat_2 = 0 } }
		set_variable = { FRA_ministry_seat_2 = FRA_ministry_appointee }
		add_to_variable = { FRA_ministry_seats_used = 1 }
	}
	else_if = {
		limit = { check_variable = { FRA_ministry_seat_3 = 0 } }
		set_variable = { FRA_ministry_seat_3 = FRA_ministry_appointee }
		add_to_variable = { FRA_ministry_seats_used = 1 }
	}
	clamp_variable = { var = FRA_ministry_seats_used min = 0 max = 3 }
}

# Expects FRA_ministry_leaver to be set beforehand.
FRA_ministry_seat_vacate = {
	if = {
		limit = { check_variable = { FRA_ministry_seat_1 = FRA_ministry_leaver } }
		set_variable = { FRA_ministry_seat_1 = 0 }
		subtract_from_variable = { FRA_ministry_seats_used = 1 }
	}
	if = {
		limit = { check_variable = { FRA_ministry_seat_2 = FRA_ministry_leaver } }
		set_variable = { FRA_ministry_seat_2 = 0 }
		subtract_from_variable = { FRA_ministry_seats_used = 1 }
	}
	if = {
		limit = { check_variable = { FRA_ministry_seat_3 = FRA_ministry_leaver } }
		set_variable = { FRA_ministry_seat_3 = 0 }
		subtract_from_variable = { FRA_ministry_seats_used = 1 }
	}
	clamp_variable = { var = FRA_ministry_seats_used min = 0 max = 3 }
}

#	THE PARLIAMENTARY DEBATE DISCOUNT
#---------------------------------------------------------------------------------------
#
# Only Persigny's half-price clause is implemented here; the Debate mechanic itself
# (Strengthen Parliament -> Parliamentary Debate, and Discussion of Amendments ->
# Debate the Emperor's Policy) is not written yet. When it is, it should price every
# political power cost through FRA_pay_debate_pp_cost so the discount applies
# automatically:
#
#	set_temp_variable = { FRA_debate_pp_cost = 50 }   # the undiscounted PP cost
#	FRA_pay_debate_pp_cost = yes
#
# and it may read `FRA_debate_pp_cost_factor` (0.5 or 1) to display a discounted price.

FRA_ministry_grant_debate_discount = {
	custom_effect_tooltip = FRA_persigny_debate_discount_tt
	hidden_effect = {
		set_country_flag = FRA_persigny_debate_discount
		FRA_refresh_debate_pp_cost_factor = yes
	}
}

FRA_refresh_debate_pp_cost_factor = {
	if = {
		limit = { FRA_has_persigny_debate_discount = yes }
		set_variable = { FRA_debate_pp_cost_factor = 0.5 }
	}
	else = { set_variable = { FRA_debate_pp_cost_factor = 1 } }
}

# Expects FRA_debate_pp_cost to hold the undiscounted cost as a positive number.
FRA_pay_debate_pp_cost = {
	hidden_effect = { FRA_refresh_debate_pp_cost_factor = yes }
	multiply_temp_variable = { FRA_debate_pp_cost = FRA_debate_pp_cost_factor }
	multiply_temp_variable = { FRA_debate_pp_cost = -1 }
	add_political_power = FRA_debate_pp_cost
}

# Victory chance = Bonapartist popularity / 3 x preparation, or x2 for an internal
# debate; clamped to the 5%-95% band demanded by the design.
FRA_calculate_debate_odds = {
	set_variable = { FRA_debate_win_chance = party_popularity@autocracy }
	if = {
		limit = { check_variable = { FRA_debate_opponent = 3 } }
		multiply_variable = { FRA_debate_win_chance = 2 }
		divide_variable = { FRA_debate_win_chance = 3 }
	}
	else = {
		divide_variable = { FRA_debate_win_chance = 3 }
		multiply_variable = { FRA_debate_win_chance = FRA_debate_preparation }
	}
	clamp_variable = { var = FRA_debate_win_chance min = 0.05 max = 0.95 }
}

# Moves 2% of popularity between the Bonapartists and the chosen opponent.
# Expects FRA_debate_swing to hold the swing towards the Bonapartists.
FRA_apply_debate_popularity_swing = {
	add_popularity = { ideology = autocracy popularity = FRA_debate_swing }
	set_temp_variable = { FRA_debate_counter_swing = FRA_debate_swing }
	multiply_temp_variable = { FRA_debate_counter_swing = -1 }
	if = {
		limit = { check_variable = { FRA_debate_opponent = 1 } }
		add_popularity = { ideology = ideology_social_liberalism popularity = FRA_debate_counter_swing }
	}
	else_if = {
		limit = { check_variable = { FRA_debate_opponent = 2 } }
		add_popularity = { ideology = centrism popularity = FRA_debate_counter_swing }
	}
}

# Picks the opposition party that heckles the Emperor, weighted by its popularity.
FRA_pick_the_heckling_party = {
	set_temp_variable = { FRA_heckle_republicans = party_popularity@ideology_social_liberalism }
	set_temp_variable = { FRA_heckle_conservative_liberals = party_popularity@centrism }
	set_temp_variable = { FRA_heckle_socialists = party_popularity@social_egalitarianism }
	set_temp_variable = { FRA_heckle_total = FRA_heckle_republicans }
	add_to_temp_variable = { FRA_heckle_total = FRA_heckle_conservative_liberals }
	add_to_temp_variable = { FRA_heckle_total = FRA_heckle_socialists }
	if = {
		limit = { check_variable = { FRA_heckle_total < 0.01 } }
		set_variable = { FRA_heckler_party = 1 }
	}
	else = {
		set_temp_variable = { FRA_heckle_roll = random }
		multiply_temp_variable = { FRA_heckle_roll = FRA_heckle_total }
		if = {
			limit = { check_variable = { FRA_heckle_roll < FRA_heckle_republicans } }
			set_variable = { FRA_heckler_party = 1 }
		}
		else = {
			subtract_from_temp_variable = { FRA_heckle_roll = FRA_heckle_republicans }
			if = {
				limit = { check_variable = { FRA_heckle_roll < FRA_heckle_conservative_liberals } }
				set_variable = { FRA_heckler_party = 2 }
			}
			else = { set_variable = { FRA_heckler_party = 3 } }
		}
	}
}

	set_temp_variable = { FRA_election_counter = FRA_election_swing }
	if = {
		limit = { check_variable = { FRA_election_outcome = 2 } }
		multiply_temp_variable = { FRA_election_counter = -0.5 }
		add_popularity = { ideology = autocracy popularity = FRA_election_shift }
		add_popularity = { ideology = ideology_social_liberalism popularity = FRA_election_counter }
		add_popularity = { ideology = centrism popularity = FRA_election_counter }
		add_stability = -0.01
	}
	else = {
		multiply_temp_variable = { FRA_election_shift = -1 }
		multiply_temp_variable = { FRA_election_counter = 0.5 }
		add_popularity = { ideology = autocracy popularity = FRA_election_shift }
		add_popularity = { ideology = ideology_social_liberalism popularity = FRA_election_counter }
		add_popularity = { ideology = centrism popularity = FRA_election_counter }
		if = {
			limit = { check_variable = { FRA_election_outcome = 1 } }
			add_stability = -0.05
		}
		else = { add_stability = -0.02 }
	}
}

# The Midi rises first: Charente, Limousin, Auvergne, the Rhone and everything south,
# plus Corsica. Algeria only joins once four or more crackdown decrees have been signed.
FRA_start_the_southern_uprising = {
	set_variable = { FRA_uprising_size = FRA_conservative_uprising_chance }
	divide_variable = { FRA_uprising_size = 100 }
	add_to_variable = { FRA_uprising_size = 0.3 }
	clamp_variable = { var = FRA_uprising_size min = 0.3 max = 0.6 }
	if = {
		limit = { FRA_algeria_joins_the_uprising = yes }
		start_civil_war = {
			ruling_party = autocracy
			ideology = ideology_social_liberalism
			size = FRA_uprising_size
			keep_political_leader = yes
			states = { 1 19 20 21 22 23 25 26 31 32 33 971 459 460 513 1013 1382 1384 1415 }
		}
	}
	else = {
		start_civil_war = {
			ruling_party = autocracy
			ideology = ideology_social_liberalism
			size = FRA_uprising_size
			keep_political_leader = yes
			states = { 1 19 20 21 22 23 25 26 31 32 33 971 }
		}
	}
	set_country_flag = FRA_southern_uprising_ongoing
	clr_country_flag = FRA_conservative_uprising_watch
	country_event = { id = france.297 days = 30 }
}
