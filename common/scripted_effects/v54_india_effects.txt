# Victorian 1854-1900 project: India. EoaNB starts in the middle of the Sepoy rebellion (May 1857); with a
# 1854 start the rebellion must be created when it breaks out. v54_start_sepoy_rebellion replays what the
# upstream start did (history/countries/SRS - Sepoy RS.txt and common/on_actions/_on_startup_events.txt).
# Scope: any country. Fired by event v54_gbr.80 when the unrest on the subcontinent has reached breaking point.
v54_start_sepoy_rebellion = {
	set_global_flag = v54_sepoy_rebellion_started
	set_global_flag = eoanb_flag_sepoy_rebellion_ongoing
	set_global_flag = eoanb_flag_sow_ongoing
	SRS = {
		transfer_state = 439
		transfer_state = 438
		transfer_state = 1128
		transfer_state = 1006
		set_capital = { state = 439 }
		load_oob = "SRS_1857"
		set_cosmetic_tag = SRS_Sepoys
		create_faction_from_template = faction_template_sepoy_rebellion
		set_country_flag = ENG_sepoy_rebellious_state_diplomacy_flag
		complete_national_focus = SRS_march_to_delhi
		complete_national_focus = SRS_pettion_to_bahadur_shad
		activate_mission = SRS_victory_sepoy_rebellion
		if = {
			limit = { country_exists = OUD }
			add_to_faction = OUD
		}
		if = {
			limit = { country_exists = GWA }
			add_to_faction = GWA
		}
		declare_war_on = { target = RAJ type = annex_everything }
		country_event = { id = sepoy.1 days = 5 }
	}
	if = {
		limit = { country_exists = KAS }
		KAS = { add_to_war = { targeted_alliance = RAJ enemy = SRS } }
	}
	if = {
		limit = { country_exists = RJP }
		RJP = { add_to_war = { targeted_alliance = RAJ enemy = SRS } }
	}
	if = {
		limit = { country_exists = KAR }
		KAR = { add_to_war = { targeted_alliance = RAJ enemy = SRS } }
	}
	if = {
		limit = { country_exists = JPR }
		JPR = { add_to_war = { targeted_alliance = RAJ enemy = SRS } }
	}
	if = {
		limit = { country_exists = GWA }
		GWA = { add_to_war = { targeted_alliance = SRS enemy = RAJ } }
	}
	if = {
		limit = { country_exists = OUD }
		OUD = { add_to_war = { targeted_alliance = SRS enemy = RAJ } }
	}
}
