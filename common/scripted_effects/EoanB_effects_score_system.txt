eoanb_score_system_calc = {
	clear_array = global.score_array
	clear_array = global.score_tag_array
	clear_array = global.score_array_position

	every_country = {
		limit = {
			NOT = {
				OR = {
					tag = UCT
					tag = AFR
				}
			}
		}
		eoanb_score_vars_init = yes			# guarantees prestige_score / tech_score exist before they are read or clamped
		eoanb_score_calc_industrial_score = yes
		eoanb_score_calc_military_score = yes
	}
	
	every_country = {
		limit = {
			is_puppet = yes
			NOT = {
				OR = {
					tag = UCT
					tag = AFR
				}
			}
		}
		OVERLORD = {
			add_to_variable = { industrial_score = PREV.industrial_score }
			add_to_variable = { military_score = PREV.military_score }
		}
	}

	every_country = {
		limit = {
			NOT = {
				OR = {
					tag = UCT
					tag = AFR
				}
			}
		}

		multiply_variable = { industrial_score = 0.5 }
		multiply_variable = { military_score = 0.5 }

		clamp_variable = {
			var = prestige_score
			min = 0
			max = 1000000
		}

		set_variable = { overall_score = industrial_score }
		add_to_variable = { overall_score = military_score }
		add_to_variable = { overall_score = prestige_score }
		add_to_variable = { overall_score = tech_score }
		clamp_variable = {
			var = overall_score
			min = 0
			max = 1000000
		}

		add_to_temp_array = { eoanb_sort_scores = overall_score }
		add_to_temp_array = { eoanb_sort_tags = THIS }
	}

	while_loop_effect = {
		limit = { check_variable = { eoanb_sort_scores^num > 0 } }
		find_highest_in_array = {
			array = eoanb_sort_scores
			value = eoanb_top_value
			index = eoanb_top_index
		}
		add_to_array = { global.score_array = eoanb_top_value }
		add_to_array = { global.score_tag_array = eoanb_sort_tags^eoanb_top_index }
		remove_from_temp_array = { array = eoanb_sort_scores index = eoanb_top_index }
		remove_from_temp_array = { array = eoanb_sort_tags index = eoanb_top_index }
	}

	for_loop_effect = {
		start = 0
		end = global.score_tag_array^num
		value = eoanb_index

		set_temp_variable = { eoanb_rank = eoanb_index }
		add_to_temp_variable = { eoanb_rank = 1 }
		add_to_array = { global.score_array_position = eoanb_rank }

		var:global.score_tag_array^eoanb_index = {
			set_variable = { score_prev_position_var = score_array_position_var }
			set_variable = { score_array_position_var = eoanb_rank }
			set_variable = { score_array_position_var_real = eoanb_index }
		}
	}
}


eoanb_ledger_system_calc = {
	clear_array = global.population_array
	clear_array = global.population_array_b
	clear_array = global.factories_array
	clear_array = global.divisions_array
	clear_array = global.ships_array

	set_temp_variable = { ledger_max_ind = 0.001 }
	set_temp_variable = { ledger_max_mil = 0.001 }
	set_temp_variable = { ledger_max_pres = 0.001 }
	set_temp_variable = { ledger_max_tech = 0.001 }

	for_loop_effect = {
		start = 0
		end = global.score_tag_array^num
		value = eoanb_index

		set_temp_variable = { eoanb_pop = 1 }
		set_temp_variable = { eoanb_pop_b = 1 }
		set_temp_variable = { eoanb_fac = 1 }
		set_temp_variable = { eoanb_div = 1 }
		set_temp_variable = { eoanb_ships = 1 }
		set_temp_variable = { eoanb_listed = 0 }

		var:global.score_tag_array^eoanb_index = {
			if = {
				limit = {
					NOT = {
						OR = {
							tag = UCT
							tag = AFR
						}
					}
				}
				set_temp_variable = { eoanb_listed = 1 }
				set_temp_variable = { eoanb_pop   = max_manpower_k }	# total population, in thousands
				set_temp_variable = { eoanb_fac   = num_of_factories }
				set_temp_variable = { eoanb_div   = num_divisions }
				set_temp_variable = { eoanb_ships = num_ships }

				eoanb_ledger_store_country_stats = yes

				if = { limit = { check_variable = { industrial_score > ledger_max_ind } } set_temp_variable = { ledger_max_ind = industrial_score } }
				if = { limit = { check_variable = { military_score > ledger_max_mil } } set_temp_variable = { ledger_max_mil = military_score } }
				if = { limit = { check_variable = { prestige_score > ledger_max_pres } } set_temp_variable = { ledger_max_pres = prestige_score } }
				if = { limit = { check_variable = { tech_score > ledger_max_tech } } set_temp_variable = { ledger_max_tech = tech_score } }
			}
		}

		if = {
			limit = { check_variable = { eoanb_listed > 0 } }
			multiply_temp_variable = { eoanb_pop = 0.001 }		# thousands -> millions
			set_temp_variable = { eoanb_pop_b = eoanb_pop }
			multiply_temp_variable = { eoanb_pop_b = 0.001 }
		}

		add_to_array = { global.population_array = eoanb_pop }
		add_to_array = { global.population_array_b = eoanb_pop_b }
		add_to_array = { global.factories_array = eoanb_fac }
		add_to_array = { global.divisions_array = eoanb_div }
		add_to_array = { global.ships_array = eoanb_ships }
	}

	set_temp_variable = { ledger_max_power = 0.001 }
	if = {
		limit = { check_variable = { global.score_array^num > 0 } }
		set_temp_variable = { ledger_max_power = global.score_array^0 }
	}
	if = {
		limit = { check_variable = { ledger_max_power < 0.001 } }
		set_temp_variable = { ledger_max_power = 0.001 }
	}

	for_each_scope_loop = {
		array = global.score_tag_array
		if = {
			limit = { exists = yes }
			eoanb_ledger_store_country_visuals = yes
		}
	}

	every_country = {
		limit = {
			is_ai = no
			has_country_flag = eoanb_ledger_initialised
			check_variable = { show_overview_info > 0 }
		}
		eoanb_ledger_build_view = yes
	}
}

eoanb_ledger_store_country_stats = {
	set_variable = { ledger_pop_k = max_manpower_k }
	set_variable = { ledger_fac = num_of_factories }
	set_variable = { ledger_civ = num_of_civilian_factories }
	set_variable = { ledger_mil = num_of_military_factories }
	set_variable = { ledger_dock = num_of_naval_factories }
	set_variable = { ledger_div = num_divisions }
	set_variable = { ledger_ships = num_ships }
	set_variable = { ledger_mp_k = manpower_k }
	set_variable = { ledger_stab = has_stability }
	set_variable = { ledger_ws = has_war_support }
}

eoanb_ledger_store_country_visuals = {
	set_variable = { ledger_f_power = overall_score }
	divide_variable = { ledger_f_power = ledger_max_power }
	multiply_variable = { ledger_f_power = 100 }
	round_variable = ledger_f_power
	clamp_variable = { var = ledger_f_power min = 0 max = 100 }

	set_variable = { ledger_f_ind = industrial_score }
	divide_variable = { ledger_f_ind = ledger_max_ind }
	multiply_variable = { ledger_f_ind = 100 }
	round_variable = ledger_f_ind
	clamp_variable = { var = ledger_f_ind min = 0 max = 100 }

	set_variable = { ledger_f_mil = military_score }
	divide_variable = { ledger_f_mil = ledger_max_mil }
	multiply_variable = { ledger_f_mil = 100 }
	round_variable = ledger_f_mil
	clamp_variable = { var = ledger_f_mil min = 0 max = 100 }

	set_variable = { ledger_f_pres = prestige_score }
	divide_variable = { ledger_f_pres = ledger_max_pres }
	multiply_variable = { ledger_f_pres = 100 }
	round_variable = ledger_f_pres
	clamp_variable = { var = ledger_f_pres min = 0 max = 100 }

	set_variable = { ledger_f_tech = tech_score }
	divide_variable = { ledger_f_tech = ledger_max_tech }
	multiply_variable = { ledger_f_tech = 100 }
	round_variable = ledger_f_tech
	clamp_variable = { var = ledger_f_tech min = 0 max = 100 }

	if = {
		limit = { check_variable = { score_array_position_var = 1 } }
		set_variable = { ledger_f_badge = 1 }
	}
	else_if = {
		limit = { check_variable = { score_array_position_var = 2 } }
		set_variable = { ledger_f_badge = 2 }
	}
	else_if = {
		limit = { check_variable = { score_array_position_var = 3 } }
		set_variable = { ledger_f_badge = 3 }
	}
	else_if = {
		limit = { check_variable = { score_array_position_var < 9 } }
		set_variable = { ledger_f_badge = 4 }
	}
	else_if = {
		limit = { check_variable = { score_array_position_var < 17 } }
		set_variable = { ledger_f_badge = 5 }
	}
	else = {
		set_variable = { ledger_f_badge = 6 }
	}

	set_variable = { ledger_move = 0 }
	if = {
		limit = { check_variable = { score_prev_position_var > 0 } }
		set_variable = { ledger_move = score_prev_position_var }
		subtract_from_variable = { ledger_move = score_array_position_var }
	}
	if = {
		limit = { check_variable = { ledger_move > 0 } }
		set_variable = { ledger_f_move = 1 }
		set_variable = { ledger_move_abs = ledger_move }
	}
	else_if = {
		limit = { check_variable = { ledger_move < 0 } }
		set_variable = { ledger_f_move = 2 }
		set_variable = { ledger_move_abs = ledger_move }
		multiply_variable = { ledger_move_abs = -1 }
	}
	else = {
		set_variable = { ledger_f_move = 3 }
		set_variable = { ledger_move_abs = 0 }
	}
}

eoanb_ledger_init = {
	if = {
		limit = { NOT = { has_country_flag = eoanb_ledger_initialised } }
		set_country_flag = eoanb_ledger_initialised
		set_variable = { ledger_tab = 0 }
		set_variable = { ledger_sort = 0 }
		set_variable = { ledger_sort_asc = 0 }
		set_variable = { ledger_filter = 0 }
		set_variable = { ledger_dirty = 1 }
	}
}

eoanb_ledger_toggle = {
	if = {
		limit = {
			check_variable = { show_overview_info > 0 }
			has_country_flag = eoanb_ledger_initialised
		}
		set_variable = { show_overview_info = 0 }
	}
	else = {
		set_variable = { show_overview_info = 1 }
		eoanb_ledger_init = yes
		eoanb_ledger_system_calc = yes
	}
}

eoanb_ledger_click_sort = {
	if = {
		limit = { check_variable = { ledger_sort = ledger_new_sort } }
		set_temp_variable = { ledger_t_flip = 1 }
		subtract_from_temp_variable = { ledger_t_flip = ledger_sort_asc }
		set_variable = { ledger_sort_asc = ledger_t_flip }
	}
	else = {
		set_variable = { ledger_sort = ledger_new_sort }
		set_variable = { ledger_sort_asc = 0 }
	}
	eoanb_ledger_build_view = yes
}

	set_variable = { ledger_filter = ledger_new_filter }
	eoanb_ledger_build_view = yes
}

	set_temp_variable = { ledger_t_filter = ledger_filter }

		array = global.score_tag_array
		if = {
			limit = {
				exists = yes
				OR = {
					check_variable = { ledger_t_filter = 0 }
					AND = {
						check_variable = { ledger_t_filter = 1 }
						check_variable = { score_array_position_var < 9 }
					}
					AND = {
						check_variable = { ledger_t_filter = 2 }
						check_variable = { score_array_position_var < 17 }
					}
					AND = {
						check_variable = { ledger_t_filter = 3 }
						check_variable = { score_array_position_var < 41 }
					}
					AND = {
						check_variable = { ledger_t_filter = 4 }
						OR = {
							tag = PREV
							is_neighbor_of = PREV
						}
					}
				}
			}
			eoanb_ledger_sort_key = yes
			add_to_temp_array = { ledger_t_keys = ledger_t_key }
			add_to_temp_array = { ledger_t_tags = THIS }
		}
	}

	set_temp_variable = { ledger_t_par = 0 }
	while_loop_effect = {
		limit = { check_variable = { ledger_t_keys^num > 0 } }

		if = {
			limit = { check_variable = { ledger_sort_asc > 0 } }
			find_lowest_in_array = { array = ledger_t_keys value = ledger_t_val index = ledger_t_idx }
		}
		else = {
			find_highest_in_array = { array = ledger_t_keys value = ledger_t_val index = ledger_t_idx }
		}
		set_temp_variable = { ledger_t_pick = ledger_t_tags^ledger_t_idx }
		remove_from_temp_array = { array = ledger_t_keys index = ledger_t_idx }
		remove_from_temp_array = { array = ledger_t_tags index = ledger_t_idx }

		add_to_temp_variable = { ledger_t_style = ledger_t_par }
		set_temp_variable = { ledger_t_par2 = 1 }
		subtract_from_temp_variable = { ledger_t_par2 = ledger_t_par }
		set_temp_variable = { ledger_t_par = ledger_t_par2 }

		var:ledger_t_pick = {
			if = {
				limit = { tag = PREV }
				set_temp_variable = { ledger_t_style = 3 }
			}
			else_if = {
				limit = { has_war_with = PREV }
				set_temp_variable = { ledger_t_style = 5 }
			}
			else_if = {
				limit = {
					OR = {
						is_in_faction_with = PREV
						is_subject_of = PREV
						PREV = { is_subject_of = PREV }
					}
				}
				set_temp_variable = { ledger_t_style = 4 }
			}
			set_temp_variable = { ledger_t_badge = ledger_f_badge }
			set_temp_variable = { ledger_t_movef = ledger_f_move }
			set_temp_variable = { ledger_t_moveabs = ledger_move_abs }
			set_temp_variable = { ledger_t_bpow = ledger_f_power }
			set_temp_variable = { ledger_t_bind = ledger_f_ind }
			set_temp_variable = { ledger_t_bmil = ledger_f_mil }
			set_temp_variable = { ledger_t_bpres = ledger_f_pres }
			set_temp_variable = { ledger_t_btech = ledger_f_tech }
		}

		add_to_array = { ledger_order = ledger_t_pick }
		add_to_array = { ledger_rowstyle = ledger_t_style }
		add_to_array = { ledger_badge = ledger_t_badge }
		add_to_array = { ledger_movef = ledger_t_movef }
		add_to_array = { ledger_moveabs = ledger_t_moveabs }
		add_to_array = { ledger_bar_power = ledger_t_bpow }
		add_to_array = { ledger_bar_ind = ledger_t_bind }
		add_to_array = { ledger_bar_mil = ledger_t_bmil }
		add_to_array = { ledger_bar_pres = ledger_t_bpres }
		add_to_array = { ledger_bar_tech = ledger_t_btech }
	}
	set_variable = { ledger_count = ledger_order^num }

	eoanb_ledger_update_controls = yes
	add_to_variable = { ledger_dirty = 1 }
}
