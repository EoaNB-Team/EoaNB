eoanb_score_system_calc = {
	clear_array = global.score_array
	clear_array = global.score_tag_array
	clear_array = global.score_array_position

	every_country = {
		limit = {
			NOT = {
				OR = {
					tag = UCT
					tag = AFR
				}
			}
		}
		eoanb_score_vars_init = yes			# guarantees prestige_score / tech_score exist before they are read or clamped
		eoanb_score_calc_industrial_score = yes
		eoanb_score_calc_military_score = yes
	}
	
	every_country = {
		limit = {
			is_puppet = yes
			NOT = {
				OR = {
					tag = UCT
					tag = AFR
				}
			}
		}
		OVERLORD = {
			add_to_variable = { industrial_score = PREV.industrial_score }
			add_to_variable = { military_score = PREV.military_score }
		}
	}

	every_country = {
		limit = {
			NOT = {
				OR = {
					tag = UCT
					tag = AFR
				}
			}
		}

		multiply_variable = { industrial_score = 0.5 }
		multiply_variable = { military_score = 0.5 }

		clamp_variable = {
			var = prestige_score
			min = 0
			max = 1000000
		}

		set_variable = { overall_score = industrial_score }
		add_to_variable = { overall_score = military_score }
		add_to_variable = { overall_score = prestige_score }
		add_to_variable = { overall_score = tech_score }
		clamp_variable = {
			var = overall_score
			min = 0
			max = 1000000
		}

		add_to_temp_array = { eoanb_sort_scores = overall_score }
		add_to_temp_array = { eoanb_sort_tags = THIS }
	}

	while_loop_effect = {
		limit = { check_variable = { eoanb_sort_scores^num > 0 } }
		find_highest_in_array = {
			array = eoanb_sort_scores
			value = eoanb_top_value
			index = eoanb_top_index
		}
		add_to_array = { global.score_array = eoanb_top_value }
		add_to_array = { global.score_tag_array = eoanb_sort_tags^eoanb_top_index }
		remove_from_temp_array = { array = eoanb_sort_scores index = eoanb_top_index }
		remove_from_temp_array = { array = eoanb_sort_tags index = eoanb_top_index }
	}

	for_loop_effect = {
		start = 0
		end = global.score_tag_array^num
		value = eoanb_index

		set_temp_variable = { eoanb_rank = eoanb_index }
		add_to_temp_variable = { eoanb_rank = 1 }
		add_to_array = { global.score_array_position = eoanb_rank }

		var:global.score_tag_array^eoanb_index = {
			set_variable = { score_array_position_var = eoanb_rank }
			set_variable = { score_array_position_var_real = eoanb_index }
		}
	}
}


eoanb_ledger_system_calc = {
	clear_array = global.population_array
	clear_array = global.population_array_b
	clear_array = global.factories_array
	clear_array = global.divisions_array
	clear_array = global.ships_array

	for_loop_effect = {
		start = 0
		end = global.score_tag_array^num
		value = eoanb_index

		set_temp_variable = { eoanb_pop = 1 }
		set_temp_variable = { eoanb_pop_b = 1 }
		set_temp_variable = { eoanb_fac = 1 }
		set_temp_variable = { eoanb_div = 1 }
		set_temp_variable = { eoanb_ships = 1 }
		set_temp_variable = { eoanb_listed = 0 }

		var:global.score_tag_array^eoanb_index = {
			if = {
				limit = {
					NOT = {
						OR = {
							tag = UCT
							tag = AFR
						}
					}
				}
				set_temp_variable = { eoanb_listed = 1 }
				set_temp_variable = { eoanb_pop   = max_manpower_k }	# total population, in thousands
				set_temp_variable = { eoanb_fac   = num_of_factories }
				set_temp_variable = { eoanb_div   = num_divisions }
				set_temp_variable = { eoanb_ships = num_ships }

			}
		}

		if = {
			limit = { check_variable = { eoanb_listed > 0 } }
			multiply_temp_variable = { eoanb_pop = 0.001 }		# thousands -> millions
			set_temp_variable = { eoanb_pop_b = eoanb_pop }
			multiply_temp_variable = { eoanb_pop_b = 0.001 }
		}

		add_to_array = { global.population_array = eoanb_pop }
		add_to_array = { global.population_array_b = eoanb_pop_b }
		add_to_array = { global.factories_array = eoanb_fac }
		add_to_array = { global.divisions_array = eoanb_div }
		add_to_array = { global.ships_array = eoanb_ships }
	}
}
