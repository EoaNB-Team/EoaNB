eoanb_literacy_init = {
	set_country_flag = eoanb_literacy_initialised

	eoanb_literacy_set_baseline_1857 = yes

	set_temp_variable = { literacy_ff_years = global.year }
	subtract_from_temp_variable = { literacy_ff_years = 1857 }
	if = {
		limit = { check_variable = { literacy_ff_years > 0 } }
		set_temp_variable = { literacy_skip_turmoil = 1 }
		for_loop_effect = {
			end = literacy_ff_years
			value = literacy_ff_i

			eoanb_literacy_calc_growth = yes
			set_temp_variable = { literacy_step = literacy_growth }
			multiply_temp_variable = { literacy_step = 0.6 }
			add_to_variable = { literacy_rate = literacy_step }
			clamp_variable = {
				var = literacy_rate
				min = 0
				max = 99
			}
		}
	}

	set_temp_variable = { literacy_skip_turmoil = 0 }
	eoanb_literacy_calc_growth = yes
	set_variable = { literacy_pop_k = max_manpower_k }
	eoanb_literacy_update_tier = yes
}

eoanb_literacy_inherit_from_prev = {
	set_country_flag = eoanb_literacy_initialised
	set_variable = { literacy_rate = PREV.literacy_rate }
	set_variable = { literacy_growth_custom = PREV.literacy_growth_custom }
	set_variable = { literacy_ceiling_custom = PREV.literacy_ceiling_custom }
	set_variable = { literacy_tier_peak = PREV.literacy_tier_peak }
	set_temp_variable = { literacy_skip_turmoil = 0 }
	eoanb_literacy_calc_growth = yes
	set_variable = { literacy_pop_k = max_manpower_k }
	eoanb_literacy_update_tier = yes
}

eoanb_literacy_calc_growth = {
	if = {
		limit = { has_idea = public_schooling_system }
		set_variable = { literacy_base_growth = 1.1 }
		set_variable = { literacy_ceiling = 99 }
		set_temp_variable = { literacy_supply_floor = 0.5 }
	}
	else_if = {
		limit = { has_idea = private_education }
		set_variable = { literacy_base_growth = 0.5 }
		set_variable = { literacy_ceiling = 80 }
		set_temp_variable = { literacy_supply_floor = 0.25 }
	}
	else_if = {
		limit = { has_idea = church_sponsored_education }
		set_variable = { literacy_base_growth = 0.35 }
		set_variable = { literacy_ceiling = 60 }
		set_temp_variable = { literacy_supply_floor = 0.25 }
	}
	else_if = {
		limit = { has_idea = traditional_ulama }
		set_variable = { literacy_base_growth = 0.25 }
		set_variable = { literacy_ceiling = 40 }
		set_temp_variable = { literacy_supply_floor = 0.25 }
	}
	else_if = {
		limit = { has_idea = oral_tradition }
		set_variable = { literacy_base_growth = 0.05 }
		set_variable = { literacy_ceiling = 15 }
		set_temp_variable = { literacy_supply_floor = 0.25 }
	}
	else = {
		set_variable = { literacy_base_growth = 0.25 }
		set_variable = { literacy_ceiling = 40 }
		set_temp_variable = { literacy_supply_floor = 0.25 }
	}
	add_to_variable = { literacy_ceiling = literacy_ceiling_custom }
	clamp_variable = {
		var = literacy_ceiling
		min = 0
		max = 99
	}

	set_variable = { literacy_supply_factor = literacy_rate }
	divide_variable = { literacy_supply_factor = 50 }
	clamp_variable = {
		var = literacy_supply_factor
		min = 0
		max = 1
	}
	set_temp_variable = { literacy_supply_span = 1 }
	subtract_from_temp_variable = { literacy_supply_span = literacy_supply_floor }
	multiply_variable = { literacy_supply_factor = literacy_supply_span }
	add_to_variable = { literacy_supply_factor = literacy_supply_floor }

	set_variable = { literacy_mod_spending = 0 }
	if = {
		limit = { has_idea = research_spending_level_1 }
		set_variable = { literacy_mod_spending = -0.2 }
	}
	else_if = {
		limit = { has_idea = research_spending_level_2 }
		set_variable = { literacy_mod_spending = -0.1 }
	}
	else_if = {
		limit = { has_idea = research_spending_level_4 }
		set_variable = { literacy_mod_spending = 0.15 }
	}
	else_if = {
		limit = { has_idea = research_spending_level_5 }
		set_variable = { literacy_mod_spending = 0.3 }
	}

	set_variable = { literacy_mod_child = 0 }
	if = {
		limit = { has_idea = legal_child_labour }
		set_variable = { literacy_mod_child = -0.1 }
	}
	else_if = {
		limit = { has_idea = banned_child_labour }
		set_variable = { literacy_mod_child = 0.1 }
	}

	set_variable = { literacy_mod_women = 0 }
	if = {
		limit = { has_idea = legal_guardianship }
		set_variable = { literacy_mod_women = -0.05 }
	}
	else_if = {
		limit = { has_idea = women_in_the_workforce }
		set_variable = { literacy_mod_women = 0.05 }
	}
	else_if = {
		limit = { has_idea = womens_suffrage }
		set_variable = { literacy_mod_women = 0.1 }
	}

	set_variable = { literacy_mod_press = 0 }
	if = {
		limit = { has_idea = no_press }
		set_variable = { literacy_mod_press = -0.1 }
	}
	else_if = {
		limit = {
			OR = {
				has_idea = state_controlled_press
				has_idea = censored_press
			}
		}
		set_variable = { literacy_mod_press = -0.05 }
	}
	else_if = {
		limit = { has_idea = free_press }
		set_variable = { literacy_mod_press = 0.05 }
	}

	set_variable = { literacy_mod_bondage = 0 }
	if = {
		limit = { has_idea = serfdom }
		add_to_variable = { literacy_mod_bondage = -0.15 }
	}
	if = {
		limit = {
			OR = {
				has_idea = slave_trade
				has_idea = slavery_legal
			}
		}
		add_to_variable = { literacy_mod_bondage = -0.1 }
	}
	else_if = {
		limit = { has_idea = debt_slavery }
		add_to_variable = { literacy_mod_bondage = -0.05 }
	}

	set_variable = { literacy_mod_discrimination = 0 }
	if = {
		limit = {
			OR = {
				has_idea = national_supremacy
				has_idea = racial_segregation
			}
		}
		set_variable = { literacy_mod_discrimination = -0.05 }
	}
	else_if = {
		limit = { has_idea = full_equality }
		set_variable = { literacy_mod_discrimination = 0.05 }
	}

	set_variable = { literacy_mod_turmoil = 0 }
	if = {
		limit = { NOT = { check_variable = { literacy_skip_turmoil = 1 } } }
		if = {
			limit = { has_war = yes }
			add_to_variable = { literacy_mod_turmoil = -0.25 }
		}
		if = {
			limit = { has_idea = bankruptcy }
			add_to_variable = { literacy_mod_turmoil = -0.5 }
		}
		if = {
			limit = { has_stability < 0.3 }
			add_to_variable = { literacy_mod_turmoil = -0.15 }
		}
	}

	set_variable = { literacy_mod_total = literacy_mod_spending }
	add_to_variable = { literacy_mod_total = literacy_mod_child }
	add_to_variable = { literacy_mod_total = literacy_mod_women }
	add_to_variable = { literacy_mod_total = literacy_mod_press }
	add_to_variable = { literacy_mod_total = literacy_mod_bondage }
	add_to_variable = { literacy_mod_total = literacy_mod_discrimination }
	add_to_variable = { literacy_mod_total = literacy_mod_turmoil }
	clamp_variable = {
		var = literacy_mod_total
		min = -0.9
		max = 1
	}

	set_temp_variable = { literacy_mod_mult = 1 }
	add_to_temp_variable = { literacy_mod_mult = literacy_mod_total }

	set_variable = { literacy_growth = literacy_base_growth }
	multiply_variable = { literacy_growth = literacy_mod_mult }
	multiply_variable = { literacy_growth = literacy_supply_factor }
	add_to_variable = { literacy_growth = literacy_growth_custom }

	set_temp_variable = { literacy_headroom = literacy_ceiling }
	subtract_from_temp_variable = { literacy_headroom = literacy_rate }
	set_variable = { literacy_ceiling_factor = 1 }
	if = {
		limit = { check_variable = { literacy_headroom < 0.001 } }
		set_variable = { literacy_ceiling_factor = 0 }
	}
	else_if = {
		limit = { check_variable = { literacy_headroom < 10 } }
		set_variable = { literacy_ceiling_factor = literacy_headroom }
		divide_variable = { literacy_ceiling_factor = 10 }
	}
	multiply_variable = { literacy_growth = literacy_ceiling_factor }
	clamp_variable = {

eoanb_literacy_update_tier = {
	set_variable = { literacy_rate_shown = literacy_rate }
	subtract_from_variable = { literacy_rate_shown = 0.499 }
	round_variable = literacy_rate_shown
	clamp_variable = {
		var = literacy_rate_shown
		min = 0
		max = 99
	}

	if = {
		limit = { eoanb_literacy_has_national_spirit = yes }
		set_variable = { literacy_tier = 6 }
		if = {
			limit = { NOT = { has_idea = eoanb_literacy_national_spirit } }
			add_ideas = eoanb_literacy_national_spirit
		}
	}
	else_if = {
		limit = { check_variable = { literacy_rate < 20 } }
		set_variable = { literacy_tier = 1 }
		if = {
			limit = { NOT = { has_idea = eoanb_literacy_mass_illiteracy } }
			add_ideas = eoanb_literacy_mass_illiteracy
		}
	}
	else_if = {
		limit = { check_variable = { literacy_rate < 45 } }
		set_variable = { literacy_tier = 2 }
		if = {
			limit = { NOT = { has_idea = eoanb_literacy_broad_illiteracy } }
			add_ideas = eoanb_literacy_broad_illiteracy
		}
	}
	else_if = {
		limit = { check_variable = { literacy_rate < 70 } }
		set_variable = { literacy_tier = 3 }
		if = {
			limit = { NOT = { has_idea = eoanb_literacy_modest_illiteracy } }
			add_ideas = eoanb_literacy_modest_illiteracy
		}
	}
	else_if = {
		limit = { check_variable = { literacy_rate < 90 } }
		set_variable = { literacy_tier = 4 }
		if = {
			limit = { NOT = { has_idea = eoanb_literacy_limited_illiteracy } }
			add_ideas = eoanb_literacy_limited_illiteracy
		}
	}
	else = {
		set_variable = { literacy_tier = 5 }
		if = {
			limit = { NOT = { has_idea = eoanb_literacy_negligible_illiteracy } }
			add_ideas = eoanb_literacy_negligible_illiteracy
		}
	}

	if = {
		limit = { check_variable = { literacy_tier < 6 } }
		if = {
			limit = {
				check_variable = { literacy_tier_peak > 0 }
				check_variable = { literacy_tier > literacy_tier_peak }
			}
			EOANBSYS_institutions_humanitive_add_two = yes
		}
		if = {
			limit = { check_variable = { literacy_tier > literacy_tier_peak } }
			set_variable = { literacy_tier_peak = literacy_tier }
		}
	}
}

eoanb_literacy_absorb_from = {
	if = {
		limit = {
			has_country_flag = eoanb_literacy_initialised
			check_variable = { literacy_pop_k > 0 }
			FROM = {
				has_country_flag = eoanb_literacy_initialised
				check_variable = { literacy_pop_k > 0 }
			}
			eoanb_literacy_shares_home_continent_with_from = yes
		}

		set_temp_variable = { literacy_absorb_total = literacy_pop_k }
		add_to_temp_variable = { literacy_absorb_total = FROM.literacy_pop_k }
		set_temp_variable = { literacy_absorb_share = FROM.literacy_pop_k }
		divide_temp_variable = { literacy_absorb_share = literacy_absorb_total }

		set_temp_variable = { literacy_absorb_delta = FROM.literacy_rate }
		subtract_from_temp_variable = { literacy_absorb_delta = literacy_rate }
		multiply_temp_variable = { literacy_absorb_delta = literacy_absorb_share }
		add_to_variable = { literacy_rate = literacy_absorb_delta }
		clamp_variable = {
			var = literacy_rate
			min = 0
			max = 99
		}

		set_variable = { literacy_pop_k = literacy_absorb_total }
		set_variable = { FROM.literacy_pop_k = 0 }

		set_temp_variable = { literacy_skip_turmoil = 0 }
		eoanb_literacy_calc_growth = yes
		eoanb_literacy_update_tier = yes
	}
}
