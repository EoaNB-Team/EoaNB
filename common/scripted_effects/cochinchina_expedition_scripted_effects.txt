#--------------------------------------------------------------------------
#	Joining the expedition
#--------------------------------------------------------------------------
cochinchina_expedition_join_effect = {
	if = {
		limit = { TAG = SPR }
		set_country_flag = SPR_cochinchina_intervention_flag
		hidden_effect = {
			set_global_flag = SPR_cochinchina_intervention_flag
			set_country_flag = SPR_cochinchina_expedition_focus_flag
		}
	}
	else_if = {
		limit = { TAG = FRA }
		set_country_flag = FRA_cochinchina_intervention_flag
		complete_national_focus = FRA_french_indochina
		hidden_effect = { set_global_flag = FRA_cochinchina_intervention_flag }
	}
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_flag = SPR_cochinchina_intervention_global_flag } }
			set_global_flag = SPR_cochinchina_intervention_global_flag
		}
		activate_mission = SPR_cochinchina_expedition_ready_for_invasion
		activate_mission = SPR_cochinchina_expedition_begin_invasion
	}
}

cochinchina_expedition_abort_effect = {
	hidden_effect = {
		every_country = {
			limit = { OR = { TAG = SPR TAG = FRA } }
			clr_country_flag = SPR_cochinchina_intervention_flag
			clr_country_flag = FRA_cochinchina_intervention_flag
			clr_country_flag = SPR_cochinchina_expedition_begin_invasion_ready_flag
			remove_mission = SPR_cochinchina_expedition_ready_for_invasion
			remove_mission = SPR_cochinchina_expedition_begin_invasion
		}
		clr_global_flag = SPR_cochinchina_intervention_global_flag
		clr_global_flag = SPR_cochinchina_intervention_flag
		clr_global_flag = FRA_cochinchina_intervention_flag
	}
}

cochinchina_expedition_tourane_landing_effect = {
	hidden_effect = {
		every_country = {
			limit = { cochinchina_expedition_is_leader_trigger = yes }
			set_province_controller = 10309
		}
		if = {
			limit = { FRA = { has_country_flag = FRA_cochinchina_intervention_flag } }
			FRA = {
				load_oob = FRA_COCHINCHINA_LANDING
				cochinchina_expedition_ai_land_campaign_effect = yes
			}
		}
		if = {
			limit = { SPR = { has_country_flag = SPR_cochinchina_intervention_flag } }
			SPR = {
				load_oob = SPR_COCHINCHINA_LANDING
				cochinchina_expedition_ai_land_campaign_effect = yes
			}
			if = {
				limit = {
					country_exists = SPH
					SPH = { is_subject_of = SPR }
				}
				SPH = { cochinchina_expedition_ai_land_campaign_effect = yes }
			}
		}
		VIN = {
			if = {
				limit = { NOT = { has_idea = VIN_cochinchina_expedition_resolve } }
				add_ideas = VIN_cochinchina_expedition_resolve
				set_country_flag = VIN_cochinchina_expedition_resolve_flag
			}
		}
		set_global_flag = SPR_cochinchina_expedition_tourane_landing
		news_event = { id = cochinchina_expedition.19 }
	}
}

cochinchina_expedition_open_joint_access_effect = {
	if = {
		limit = {
			SPR = { has_country_flag = SPR_cochinchina_intervention_flag }
			FRA = { has_country_flag = FRA_cochinchina_intervention_flag }
		}
		SPR = {
			if = {
				limit = { NOT = { has_military_access_to = FRA } }
				diplomatic_relation = { country = FRA relation = military_access active = yes }
				set_country_flag = cochinchina_expedition_opened_access_flag
			}
			if = {
				limit = { NOT = { has_non_aggression_pact_with = FRA } }
				diplomatic_relation = { country = FRA relation = non_aggression_pact active = yes }
				set_country_flag = cochinchina_expedition_opened_nap_flag
			}
		}
		FRA = {
			if = {
				limit = { NOT = { has_military_access_to = SPR } }
				diplomatic_relation = { country = SPR relation = military_access active = yes }
				set_country_flag = cochinchina_expedition_opened_access_flag
			}
		}
		if = {
			limit = {
				country_exists = SPH
				SPH = { is_subject_of = SPR }
			}
			SPH = {
				if = {
					limit = { NOT = { has_military_access_to = FRA } }
					diplomatic_relation = { country = FRA relation = military_access active = yes }
					set_country_flag = cochinchina_expedition_opened_access_flag
				}
				if = {
					limit = { NOT = { has_non_aggression_pact_with = FRA } }
					diplomatic_relation = { country = FRA relation = non_aggression_pact active = yes }
					set_country_flag = cochinchina_expedition_opened_nap_flag
				}
			}
			FRA = {

	}
}

cochinchina_expedition_end_war_effect = {
	every_country = {
		limit = { cochinchina_expedition_is_participant_or_subject_trigger = yes }
		every_country = {
			limit = {
				OR = {
					TAG = VIN
					is_subject_of = VIN
				}
			}
			white_peace = PREV
		}
	}
}

cochinchina_expedition_capitulation_effect = {
	set_global_flag = SPR_cochinchina_expedition_capitulation_flag
	cochinchina_expedition_end_war_effect = yes
	every_country = {
		limit = { cochinchina_expedition_is_participant_trigger = yes }
		remove_mission = SPR_cochinchina_expedition_take_port_mission
		country_event = { id = cochinchina_expedition.364 }
	}
	VIN = { remove_mission = SPR_cochinchina_expedition_take_port_mission }
}

			SPR = { has_country_flag = SPR_cochinchina_intervention_flag }
			FRA = { has_country_flag = FRA_cochinchina_intervention_flag }
		}
		CCH = {
			set_cosmetic_tag = FRA_cochin_protect
			FRA = {
				set_autonomy = {
					target = PREV
					autonomy_state = autonomy_colony
				}
			}
		}
		if = {
			limit = {
				SPR = { has_completed_focus = SPR_spanish_victory_cochinchina }
				country_exists = SPH
			}
			SPH = { transfer_state = 998 }
		}
		else = {
			CCH = { transfer_state = 998 }
		}
	}
	else_if = {
		limit = { SPR = { has_country_flag = SPR_cochinchina_intervention_flag } }
		CCH = {
			transfer_state = 998
			set_cosmetic_tag = SPR_cochin_protect
			SPR = {
				set_autonomy = {
					target = PREV
					autonomy_state = autonomy_colony
				}
			}
		}
	}
	else_if = {
		limit = { FRA = { has_country_flag = FRA_cochinchina_intervention_flag } }

			transfer_state = 998
			set_cosmetic_tag = FRA_cochin_protect
			FRA = {
				set_autonomy = {
					target = PREV
					autonomy_state = autonomy_colony
				}
			}
		}
	}
	if = {

			FRA = { has_country_flag = FRA_cochinchina_intervention_flag }
			cochinchina_expedition_france_kept_to_history_trigger = yes
		}
