# This file holds the whole reform voting pipeline of the new parliament system:
#
#  1. parliament_enact_reform (in _Parliament_scripted_effects.txt) saves the clicked
#     reform token into current_reform_token and fires parliament_event.10
#  2. parliament_event.10 (immediate) calls, in this order:
#       parliament_set_reform_vote_type    -> sets vote_issue_id from the reform category
#       parliament_set_reform_majority     -> sets parliament_reform_majority (1 = simple, 2 = super)
#       parliament_set_reform_approvals    -> sets parl_issue_approval_array for the reform
#       compute_voting_approval            -> seeds parl_voting_approval_array (party values + SoG)
#       compute_number_seats_tot           -> parl_tot_num_seats / parl_num_seats_half
#       compute_number_seats_approve_est   -> parl_est_num_seats (estimate for the event text)
#  3. The vote resolves either through the parliament_voting_mission timeout or the urgent
#     option of parliament_event.10; both end in parliament_calculate_vote_result, which
#     sets the result flags and fires parliament_event.12
#  4. parliament_event.12 calls parliament_enact_voted_reform on a successful vote and
#     parliament_reform_vote_cleanup in every outcome
#  5. parliament_show_voting_decisions / parliament_hide_voting_decisions put the vote
#     swaying decisions into gui category 2 while a vote is scheduled and clear them after
#
# Party order used by every array in this file (index -> party):
#   0 rad_soc   1 soc_lib   2 soc_ega   3 cen   4 tra_con
#   5 rad_dem   6 chauv_pop   7 aut   8 fun
#
# Stance order used by parl_issue_approval_array (index -> stance):
#   0 = stance 1 (Reactionary / Laissez-Faire / Paternalist / Jingoist)
#   1 = stance 2 (Conservative / Interventionist / Elitist / Pro-Militarist)
#   2 = stance 3 (Liberal / State Capitalist / Favoritist / Anti-Militarist)
#   3 = stance 4 (Progressive / Planned Economy / Egalitarian / Pacifist)
# ideology_<party>_issues stores stances as 1-4, so every lookup subtracts 1.

# _narrow means the neighbouring stance dislikes it too.

#   parl_ap_stance_1               0.3        0.1       -0.1       -0.3
#   parl_ap_stance_2_narrow        0.1        0.3       -0.1       -0.3

# Sets vote_issue_id based on which reform category array holds current_reform_token
# Category -> issue mapping ported from the old per-reform events:
# cul: 3 (church and state), 7 (education)
# eco: 6 (health), 15 (child labour), 16 (economic structure), 17 (trade policy),
#      18 (economic mobilisation), 19 (land ownership), 20 (working hours)
# war: 5 (policing), 8 (conscription)
# dem: everything else (1, 2, 4, 9, 10, 11, 12, 13, 14)
parliament_set_reform_vote_type = {
	if = {
		limit = {
			OR = {
				is_in_array = { array = parliament_reform_array_3 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_7 value = current_reform_token }
			}
		}
		set_vote_cul = yes
	}
	else_if = {
		limit = {
			OR = {
				is_in_array = { array = parliament_reform_array_6 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_15 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_16 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_17 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_18 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_19 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_20 value = current_reform_token }
			}
		}
		set_vote_eco = yes
	}
	else_if = {
		limit = {
			OR = {
				is_in_array = { array = parliament_reform_array_5 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_8 value = current_reform_token }
			}
		}
		set_vote_war = yes
	}
	else = {
		set_vote_dem = yes
	}
}

# Sets the level of majority needed to pass the current reform
# 1 = simple majority (more than 50 percent of all seats)
# 2 = super majority (at least 65 percent of all seats)
# Super majority currently applies to categories 1 (DoP), 2 (discriminatory laws),
# 3 (church and state), 12 (rights of women) and 14 (slavery) - adjust the list to taste
parliament_set_reform_majority = {
	# Simple majority is the default.
	set_variable = { parliament_reform_majority = 1 }

	# Constitutional and fundamental-rights reforms require a 65% supermajority.
	if = {
		limit = {
			OR = {
				is_in_array = { array = parliament_reform_array_1 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_2 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_3 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_12 value = current_reform_token }
				is_in_array = { array = parliament_reform_array_14 value = current_reform_token }
			}
		}
		set_variable = { parliament_reform_majority = 2 }
	}
}

# Seeds parl_voting_approval_array with the initial support of every party for the current reform
# Formula per party:
#   0.30 base
# + 0.40 * <party>_approval_of_government (SoG, 0..1 - see the coalition scripted effects)
# + issue alignment from parl_issue_approval_array (-0.5 .. +0.3)
# + random -0.10 .. +0.10 (skipped when parl_vote_deterministic is set)
# clamped to 0..1
compute_voting_approval = {
	clear_array = parl_voting_approval_array
	resize_array = {
		array = parl_voting_approval_array
		value = 0
		size = 9
	}

	# Support of government of every party, in party order.
	clear_temp_array = parl_sog_array
	add_to_temp_array = { parl_sog_array = rad_soc_approval_of_government }
	add_to_temp_array = { parl_sog_array = soc_lib_approval_of_government }
	add_to_temp_array = { parl_sog_array = soc_ega_approval_of_government }
	add_to_temp_array = { parl_sog_array = cen_approval_of_government }
	add_to_temp_array = { parl_sog_array = tra_con_approval_of_government }
	add_to_temp_array = { parl_sog_array = rad_dem_approval_of_government }
	add_to_temp_array = { parl_sog_array = chauv_pop_approval_of_government }
	add_to_temp_array = { parl_sog_array = aut_approval_of_government }
	add_to_temp_array = { parl_sog_array = fun_approval_of_government }

	# Stance (1-4) every party takes on the issue this vote is about, in the same order.
	clear_temp_array = parl_stance_array
	add_to_temp_array = { parl_stance_array = ideology_rad_soc_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_soc_lib_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_soc_ega_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_cen_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_tra_con_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_rad_dem_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_chauv_pop_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_aut_issues^vote_issue_id }
	add_to_temp_array = { parl_stance_array = ideology_fun_issues^vote_issue_id }

	# Width of the random swing: 0.2 gives -0.1 .. +0.1, 0 makes the vote deterministic.
	set_temp_variable = { parl_rand_spread = 0.2 }
	if = {
		limit = { has_country_flag = parl_vote_deterministic }
		set_temp_variable = { parl_rand_spread = 0 }
		clr_country_flag = parl_vote_deterministic
	}

	for_each_loop = {
		array = parl_sog_array
		value = parl_sog
		index = parl_i

		# Stance 1-4 -> parl_issue_approval_array index 0-3.
		set_temp_variable = { parl_stance_i = { value = parl_stance_array^parl_i subtract = 1 } }

		# random is 0..1, so this is -0.1 .. +0.1 at the default spread and 0 for a scripted vote.
		set_temp_variable = { parl_r = { value = random subtract = 0.5 multiply = parl_rand_spread } }

		set_variable = {
			parl_voting_approval_array^parl_i = {
				value = parl_sog
				multiply = 0.40
				add = 0.30
				add = parl_issue_approval_array^parl_stance_i
				add = parl_r
				clamp = { min = 0 max = 1 }
			}
		}
	}
}

# Total number of seats in the parliament (also computes the half point)
compute_number_seats_tot = {
	set_variable = { parl_tot_num_seats = 0 }

	for_each_loop = {
		array = seats_count_array
		value = parl_party_seats

		add_to_variable = { parl_tot_num_seats = parl_party_seats }
	}

	set_variable = { parl_num_seats_half = { value = parl_tot_num_seats divide = 2 round = yes } }
}

# Estimated number of seats that would approve the reform right now (event text / tooltips)
compute_number_seats_approve_est = {
	set_variable = { parl_est_num_seats = 0 }

	for_each_loop = {
		array = seats_count_array
		value = parl_party_seats
		index = parl_i

		add_to_variable = {
			parl_est_num_seats = { value = parl_party_seats multiply = parl_voting_approval_array^parl_i }
		}
	}

	round_variable = parl_est_num_seats
}

# Rolls the vote of every seat, checks the result against the required majority and fires the
# result event. Called by the parliament_voting_mission timeout_effect and by parliament_event.11
# (the urgent vote path). Requires parliament_reform_majority to be set (see
# parliament_set_reform_majority); an unset value falls back to a simple majority.
parliament_calculate_vote_result = {
	# Prevent result flags from an interrupted or previously unresolved vote
	# from activating the wrong event option.
	clr_country_flag = eoanb_reform_vote_success
	clr_country_flag = eoanb_reform_vote_failure
	clr_country_flag = eoanb_reform_vote_tie

	set_variable = { yay_votes = 0 }
	set_variable = { nay_votes = 0 }

	# Every seat rolls against the approval of the party that holds it.
	for_each_loop = {
		array = seats_array
		value = parl_seat_party

		set_temp_variable = { parl_r = random }

		if = {
			limit = {
				check_variable = {
					var = parl_r
					value = parl_voting_approval_array^parl_seat_party
					compare = less_than
				}
			}
			add_to_variable = { yay_votes = 1 }
		}
		else = {
			add_to_variable = { nay_votes = 1 }
		}
	}

	compute_number_seats_tot = yes

	# Threshold in seats: 65% of the whole parliament for a super majority, half of it otherwise.
	set_variable = {
		parl_votes_threshold = {
			value = parl_tot_num_seats
			if = {
				limit = { value = parliament_reform_majority equals = 2 }
				multiply = 0.65
			}
			else = { multiply = 0.50 }
		}
	}

	# Convert the fractional threshold into the smallest whole number of votes that passes it.
	# Example: 65% of 256 is 166.4, so 167 votes are required.
	# A simple majority needs one extra vote when the threshold is a whole number, since exactly
	# half of the seats is not a majority: 128 of 256 seats ties, 129 wins.
	set_variable = { parl_votes_required = { value = parl_votes_threshold round = yes } }
	if = {
		limit = {
			OR = {
				check_variable = {
					var = parl_votes_required
					value = parl_votes_threshold
					compare = less_than
				}
				AND = {
					NOT = { check_variable = { parliament_reform_majority = 2 } }
					check_variable = {
						var = parl_votes_required
						value = parl_votes_threshold
						compare = equals
					}
				}
			}
		}
		add_to_variable = { parl_votes_required = 1 }
	}

	if = {
		limit = {
			check_variable = {
				var = yay_votes
				value = parl_votes_required
				compare = greater_than_or_equals
			}
		}
		set_country_flag = eoanb_reform_vote_success
	}
	else_if = {
		# Only a simple majority can end in a tie - a super majority proposal that falls short
		# simply fails, even when the yays outnumber the nays.
		limit = {
			NOT = { check_variable = { parliament_reform_majority = 2 } }
			check_variable = {
				var = yay_votes
				value = nay_votes
				compare = equals
			}
		}
		set_country_flag = eoanb_reform_vote_tie
	}
	else = {
		set_country_flag = eoanb_reform_vote_failure
	}

	hidden_effect = {
		country_event = { id = parliament_event.12 }
	}
}

# Adds the national spirit of the reform stored in current_reform_token.
# Every reform token is the token of the idea with the same name, so the idea name is pasted
# straight out of the variable with a meta effect instead of a token -> add_ideas chain.
# The reform ideas are law-type ideas, so add_ideas replaces the previous law of the same slot.
# Set debug = yes inside the meta effect to print the built effect to game.log.
parliament_enact_voted_reform = {
	if = {
		limit = {
			has_variable = current_reform_token
			check_variable = { current_reform_token > 0 }
		}
		meta_effect = {
			text = {
				add_ideas = [REFORM_IDEA]
			}
			REFORM_IDEA = "[?current_reform_token.GetTokenKey]"
			debug = no
		}
	}
	else = {
		log = "parliament_enact_voted_reform: current_reform_token is not set"
	}
}

# Clears the vote state after the result event - called by every option of parliament_event.12
# NOTE: unlock of the reform (clicked@var:current_reform_token = 0) is handled by the failure and
# tie options of the event BEFORE this runs, since this effect clears current_reform_token
parliament_reform_vote_cleanup = {
	clr_country_flag = eoanb_reform_vote_success
	clr_country_flag = eoanb_reform_vote_failure
	clr_country_flag = eoanb_reform_vote_tie
	clr_country_flag = parliament_reform_vote_ongoing
	clear_variable = parl_votes_threshold
	clear_variable = parl_votes_required
	clear_variable = current_reform_token

	# Reset the mission GUI row and all custom timer values.
	set_temp_variable = { parliament_decision_token = token:parliament_voting_mission }
	set_variable = { clicked@var:parliament_decision_token = 0 }
	clear_variable = parliament_vote_days_left
	clear_variable = parliament_vote_total_days
	clear_variable = parliament_vote_progress

	# remove the voting mission and the vote swaying decisions from the parliament gui
	# both calls are safe no-ops if the urgent path was taken and the mission never started
	if = {
		limit = { has_active_mission = parliament_voting_mission }
		remove_mission = parliament_voting_mission
	}
	parliament_hide_voting_decisions = yes
}

# Prussian Reform Missions Cleanup
prussian_voting_mission_cleanup = {
	clr_country_flag = parliament_reform_vote_ongoing

	if = {
		limit = {
			has_active_mission = prussian_voting_mission
		}
		remove_mission = prussian_voting_mission
	}

	if = {
		limit = {
			has_country_flag = PRS_expand_voting_rights_flag
		}
		set_temp_variable = { parliament_decision_token = token:PRS_expand_voting_rights }
	}

	parliament_hide_decision = yes
	parliament_reform_vote_cleanup = yes
	parliament_recompute_coalition_seats = yes

	if = {
		limit = {
			has_country_flag = PRS_expand_voting_rights_flag
		}
		clr_country_flag = PRS_expand_voting_rights_flag
	}
}
