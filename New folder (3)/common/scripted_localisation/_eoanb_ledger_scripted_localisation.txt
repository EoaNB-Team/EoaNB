defined_text = {
	name = GetLedgerStatus
	text = {
		trigger = { check_variable = { score_array_position_var < 1 } }
		localization_key = LEDGER_STATUS_UNRANKED
	}
	text = {
		trigger = {
			check_variable = { score_array_position_var < 4 }

		}
		localization_key = LEDGER_STATUS_SUPER
	}
	text = {
		trigger = { check_variable = { score_array_position_var < 9 } }
		localization_key = LEDGER_STATUS_GREAT
	}
	text = {
		trigger = { check_variable = { score_array_position_var < 17 } }
		localization_key = LEDGER_STATUS_SECONDARY
	}
	text = {
		trigger = { check_variable = { score_array_position_var < 41 } }
		localization_key = LEDGER_STATUS_MINOR
	}
	text = {
		localization_key = LEDGER_STATUS_INSIGNIFICANT
	}
}

	name = GetLedgerPopulation

			check_variable = { ledger_pop_k > 999 }

			divide_temp_variable = { ledger_t_pop_m = 1000 }
		}
		localization_key = LEDGER_POP_MILLIONS

		trigger = {
			set_temp_variable = { ledger_t_pop_k = ledger_pop_k }
		}
		localization_key = LEDGER_POP_THOUSANDS

defined_text = {
	name = GetLedgerMovement
	text = {
		trigger = { check_variable = { score_prev_position_var < 1 } }
		localization_key = LEDGER_MOVE_NEW_TT
	}
	text = {
		trigger = {
			check_variable = { ledger_move > 0 }
			set_temp_variable = { ledger_t_mv = ledger_move }
		}
		localization_key = LEDGER_MOVE_UP_TT
	}
	text = {
		trigger = {
			check_variable = { ledger_move < 0 }
			set_temp_variable = { ledger_t_mv = ledger_move }
			multiply_temp_variable = { ledger_t_mv = -1 }
		}
		localization_key = LEDGER_MOVE_DOWN_TT
	}
	text = {
		localization_key = LEDGER_MOVE_SAME_TT
	}
}
