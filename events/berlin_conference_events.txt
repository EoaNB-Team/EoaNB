# c. October 1884
country_event = {
	id = berlin_conference.100
	hidden = yes

	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		if = {
			limit = { country_exists = PRS }
			PRS = { save_global_event_target_as = berlin_conference_host }
			set_global_flag = berlin_conference_in_berlin
		}
		else_if = {
			limit = { country_exists = FRA }
			FRA = { save_global_event_target_as = berlin_conference_host }
		}
		else = {
			ENG = { save_global_event_target_as = berlin_conference_host }
		}
		set_global_flag = berlin_conference_called
		event_target:berlin_conference_host = {
			set_country_flag = berlin_conference_participant
			country_event = { id = berlin_conference.6 }
		}
	}
}

# October 1884
country_event = {
	id = berlin_conference.6
	title = berlin_conference.6.t
	desc = {
		text = berlin_conference.6.d_germany
		trigger = {
			tag = PRS
			trigger_is_germany = yes
		}
	}
	desc = {
		text = berlin_conference.6.d_prussia
		trigger = {
			tag = PRS
			NOT = { trigger_is_germany = yes }
		}
	}
	desc = {
		text = berlin_conference.6.d_other
		trigger = { NOT = { tag = PRS } }
	}
	picture = GFX_report_event_generic_conference

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = berlin_conference.6.a
		set_temp_variable = { eoanb_prestige_change = 3 }
		eoanb_change_prestige_score = yes
		hidden_effect = {
			every_country = {
				limit = {
					exists = yes
					NOT = { tag = ROOT }
					OR = {
						tag = PRS
						tag = FRA
						tag = ENG
						tag = AUS
						tag = BEL
						tag = DEN
						tag = HOL
						tag = POR
						tag = RUS
						tag = SPR
						tag = SWE
						tag = OTO
						tag = USA
						has_country_flag = is_literally_italy
					}
				}
				country_event = { id = berlin_conference.7 days = 2 }
			}
			if = {
				limit = {
					NOT = { any_country = { has_country_flag = is_literally_italy } }
					country_exists = PIE
					NOT = { tag = PIE }
				}
				PIE = { country_event = { id = berlin_conference.7 days = 2 } }
			}
		}
	}
}

country_event = {
	id = berlin_conference.102
	hidden = yes

	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		set_global_flag = berlin_conference_act_signed
		set_global_flag = scramble_for_africa
		set_global_flag = berlin_conferance_happened
		every_country = {
			limit = { has_country_flag = berlin_conference_participant }
			set_country_flag = berlin_conference_signatory
		}
		every_country = {
			limit = {
				exists = yes
				OR = {
					tag = ENG
					tag = FRA
					tag = BEL
					tag = POR
					tag = SPR
					has_country_flag = is_literally_italy
					AND = {
						tag = PRS
						trigger_is_germany = yes
					}
				}
			}
			set_country_flag = is_historical_colonial_power
		}
		every_country = {
			limit = { is_ai = no }
			set_variable = { super_event_displayed = 2 }
		}
		news_event = { id = berlin_conference.15 }
	}
}
