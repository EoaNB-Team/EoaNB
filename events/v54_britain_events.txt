	id = v54_gbr.80
	hidden = yes
	fire_only_once = yes

	mean_time_to_happen = {
		days = 1
		modifier = { factor = 0 date < 1857.5.10 }
	}

	trigger = { original_tag = ENG }

	immediate = {
		if = {
			limit = { check_variable = { v54_india_unrest > 49 } }
			v54_start_sepoy_rebellion = yes
			set_country_flag = v54_gbr_mutiny_became_rebellion
			country_event = { id = v54_gbr.81 }
		}
		else = {
			set_global_flag = v54_mutiny_contained
			country_event = { id = v54_gbr.82 }
		}
	}
}
