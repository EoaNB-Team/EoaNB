#Austria withdraws from the Papal States (premature)
country_event = {
	id = event_papacy.302
	title = event_papacy.302.t
	desc = event_papacy.302.d
	picture = GFX_report_event_generic_pope_vatican
	
	fire_only_once = yes
	
	trigger = {
		has_country_flag = AUS_papal_garrison_active
		is_ai = no
		country_exists = PAP
		NOT = { PAP_milan_has_fallen_trigger = yes }
		PAP_no_austrian_garrison_in_romagna_trigger = yes
	}
	
	mean_time_to_happen = { days = 5 }
	
	option = {
		name = event_papacy.302.a
		ai_chance = { base = 100 }
		clr_country_flag = AUS_papal_garrison_active
		set_temp_variable = { eoanb_prestige_change = -1 }
		eoanb_change_prestige_score = yes
		add_stability = -0.05
		PAP = {
			EOANBSYS_national_crisis_progress_add_ten = yes
			EOANBSYS_national_crisis_risorgimento_crisis_update = yes
			PAP_austrian_garrison_withdrawn = yes
		}
	}
}
