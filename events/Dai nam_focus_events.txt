	# Loops itself every 15-20 days, nudging the balance toward
	# Western Ideals by a random 1-3%. Started by
	# VIN_Western_Influence_Drift_Start.
	country_event = {
		id = VIN_western_influence.1
		hidden = yes

		is_triggered_only = yes

		trigger = { tag = VIN }

		immediate = {
			random_list = {
				40 = { set_temp_variable = { VIN_WI_change = 0.01 } }
				40 = { set_temp_variable = { VIN_WI_change = 0.02 } }
				20 = { set_temp_variable = { VIN_WI_change = 0.03 } }
			}
			hidden_effect = { VIN_Western_Influence_Modify = yes }
			country_event = { id = VIN_western_influence.1 days = 15 random_days = 5 }
		}
	}
