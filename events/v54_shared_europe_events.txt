# 1 - Hidden: the Russo-Turkish war is under way on 1 January 1854 (scope: Russia)
country_event = {
	id = v54_eur.1
	hidden = yes
	is_triggered_only = yes

	trigger = {
		original_tag = RUS
		v54_scenario_active = yes
		NOT = { has_global_flag = v54_crimean_war_active }
		NOT = { has_global_flag = v54_crimean_war_ended }
	}

	immediate = {
		if = {
			limit = {
				country_exists = OTO
				NOT = { has_war_with = OTO }
			}
			declare_war_on = {
				target = OTO
				type = v54_wargoal_eastern_question
			}
			activate_mission = v54_mission_crimean_peace
			v54_set_crimean_war_started = yes
			if = {
				limit = { country_exists = AUS }
				AUS = { country_event = { id = v54_eur.10 days = 20 } }
			}
			if = {
				limit = { country_exists = PRS }
				PRS = { country_event = { id = v54_eur.15 days = 30 } }
			}
			if = {
				limit = { country_exists = ENG }
				ENG = { country_event = { id = v54_eur.2 days = 60 } }
			}
			if = {
				limit = { country_exists = FRA }
				FRA = { country_event = { id = v54_eur.3 days = 63 } }
			}
			if = {
				limit = { country_exists = PIE }
				PIE = { country_event = { id = v54_eur.20 days = 330 } }
			}
		}
	}

	option = {
		name = v54_eur.1.a
	}
}

					}
				}
				country_event = { id = v54_eur.5 }

		}
	}

		name = v54_eur.2.b
